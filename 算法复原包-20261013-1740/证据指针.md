## 怎么从队列重新导出

在任意有 Node 的目录里跑这段（只读队列，不动源）：

```js
// 重新导出：按会话聚合"含指定项目路径"的原始记录
import fs from "node:fs";
import path from "node:path";
const SRC = path.join(process.env.USERPROFILE, ".openviking", "pending");
const OUT = "导出目录";
const KEY = "dshQ群小助手配置端";
fs.mkdirSync(OUT, { recursive: true });
const groups = new Map();
for (const f of fs.readdirSync(SRC).filter(x => x.endsWith(".json"))) {
  const t = fs.readFileSync(path.join(SRC, f), "utf8");
  if (!t.includes(KEY)) continue;
  let sid = "unknown", ts = 0;
  try { const o = JSON.parse(t); sid = o.sessionId || sid; ts = o.createdAt || 0; } catch {}
  if (!groups.has(sid)) groups.set(sid, []);
  groups.get(sid).push({ ts, t });
}
for (const [sid, arr] of groups) {
  arr.sort((a, b) => a.ts - b.ts);
  fs.writeFileSync(path.join(OUT, sid.replace(/[^\w.-]/g, "_") + ".jsonl"), arr.map(x => x.t.trim()).join("\n") + "\n");
}
console.log("会话数:", groups.size);
```
