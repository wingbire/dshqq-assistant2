14| function Run-Node([string[]]$Arguments, [int]$TimeoutSec) {
15|   # 直接调用（不用 Start-Process）—— 本机 PowerShell 里 Start-Process -PassThru 的 ExitCode 取不到，

18|   $watch = Start-Job -ScriptBlock {
19|     param($sec)
20|     Start-Sleep -Seconds $sec
21|     Get-CimInstance Win32_Process |
22|       Where-Object { $_.Name -eq "node.exe" -and ($_.CommandLine -like "*_ui*" -or $_.CommandLine -like "*uitest*" -or $_.CommandLine -like "*behavior-test*") } |
23|       ForEach-Object { Stop-Process -Id $_.ProcessId -Force -ErrorAction SilentlyContinue }
24|   } -ArgumentList $TimeoutSec
25|   # 【必须】用 Write-Host 打到控制台，不能让它进函数返回值 ——
26|   # 否则调用处 ((Run-Node ...) -eq 0) 会把整段输出当成返回值比较，既看不见输出、判定也永远是 false。
27|   & node @Arguments 2>&1 | ForEach-Object { Write-Host $_ }
28|   $code = $LASTEXITCODE
29|   if ($watch) { Stop-Job $watch -ErrorAction SilentlyContinue; Remove-Job $watch -Force -ErrorAction SilentlyContinue }
30|   if ($null -eq $code) { $code = 1 }
