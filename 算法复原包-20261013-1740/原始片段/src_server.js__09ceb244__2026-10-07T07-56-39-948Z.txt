function sendJson(res, code, obj) {
const body = JSON.stringify(obj);
res.writeHead(code, { "content-type": "application/json; charset=utf-8", "content-length": Buffer.byteLength(body) });
res.end(body);
}

function readBody(req) {
return new Promise(function (resolve, reject) {
let data = "";
req.on("data", function (c) { data += c; if (data.length > 2e6) { req.destroy(); reject(new Error("body too large")); } });
req.on("end", function () { resolve(data); });
req.on("error", reject);
});
}

function serveFeedQr(ctx, res, url) {
const feed = (ctx.config && ctx.config.feed) || {};
let src = String(feed.imagePath || "").trim();
// 自动抓取的二维码存在 feed 目录里，用 ?f=文件名 指定；只允许纯文件名，防目录穿越
const f = url && url.searchParams ? url.searchParams.get("f") : null;
if (f) {
const safe = path.basename(String(f));
if (safe && /\.(png|jpe?g|gif|webp)$/i.test(safe)) {
const cand = path.join(FEED_DIR, safe);
try { if (fs.existsSync(cand) && fs.statSync(cand).isFile()) src = cand; } catch (e) {}
}
}
if (!feed.enabled || !src || src.indexOf("http") === 0) {
res.writeHead(404, { "content-type": "text/plain; charset=utf-8" });
return res.end("no feed image");
}
try {
if (!fs.existsSync(src) || !fs.statSync(src).isFile()) {
res.writeHead(404); return res.end("file missing");
}
const ext = path.extname(src).toLowerCase();
res.writeHead(200, { "content-type": MIME[ext] || "image/png" });
return fs.createReadStream(src).pipe(res);
} catch (e) {
res.writeHead(500); return res.end("error");
}
}

export function createServer(ctx) {
const publicDir = path.join(APP_ROOT, "public");
return http.createServer(async function (req, res) {
const url = new URL(req.url, "http://" + (req.headers.host || "127.0.0.1"));
try {
if (url.pathname === "/feed-qr") return serveFeedQr(ctx, res, url);
if (url.pathname.indexOf("/api/") === 0) return await handleApi(ctx, req, res, url);
if (req.method !== "GET" && req.method !== "HEAD") { res.writeHead(405); return res.end(); }
const rel = url.pathname === "/" ? "index.html" : url.pathname.replace(/^[\/]+/, "");
const file = path.resolve(publicDir, rel);
if (file.indexOf(publicDir) !== 0 || !fs.existsSync(file) || !fs.statSync(file).isFile()) {
res.writeHead(404, { "content-type": "text/plain; charset=utf-8" });
return res.end("404");
}
res.writeHead(200, { "content-type": MIME[path.extname(file).toLowerCase()] || "application/octet-stream" });
return fs.createReadStream(file).pipe(res);
} catch (err) {
sendJson(res, 500, { ok: false, error: String((err && err.message) || err) });
}
});
}
