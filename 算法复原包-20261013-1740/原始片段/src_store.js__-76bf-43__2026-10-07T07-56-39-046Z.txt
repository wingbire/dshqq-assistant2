import fs from "node:fs";
import path from "node:path";
import { DATA_ROOT, ensureDir } from "./paths.js";
import { deepMerge } from "./config.js";
import { applyBand, AUTOJOIN_KEYS } from "./profile.js";

export const DATA_DIR = path.join(DATA_ROOT, "data");
export const RULES_FILE = path.join(DATA_DIR, "rules.json");
export const AUDIT_DIR = path.join(DATA_DIR, "audit");
export const GROUP_DIR = path.join(DATA_DIR, "groups");
export const CACHE_FILE = path.join(DATA_DIR, "reply-cache.json");
export const GROUP_INDEX_FILE = path.join(DATA_DIR, "groups-index.json");

export const DEFAULT_RULES = {
version: 1,
global: {
enabled: true,
aiEnabled: true,        // 本群是否参与聊天（面板「停用本群」会同时关掉它）
bannedWords: [],
bannedRegex: [],
links: { enabled: false, action: "recall" },
flood: { enabled: false, count: 5, seconds: 10, action: "mute", muteSeconds: 60 },
atAll: { enabled: false, maxPerHour: 2, action: "mute", muteSeconds: 300 },
welcome: { enabled: false, text: "欢迎 {at} 加入本群～" },
ladder: { warnLimit: 2, muteSeconds: 300, kickAfter: 0 },
autoReplies: []
},
groups: {}
};

function today() {
const d = new Date();
const p = (n) => (n < 10 ? "0" + n : "" + n);
return d.getFullYear() + "-" + p(d.getMonth() + 1) + "-" + p(d.getDate());
}

export function createStore() {
ensureDir(DATA_DIR);
ensureDir(AUDIT_DIR);
ensureDir(GROUP_DIR);

let rules = loadRules();
let cache = loadCache();
let groupIndex = loadGroupIndex();
let indexDirty = false;
let indexTimer = null;
const auditRing = [];
const MAX_AUDIT = 500;

function loadCache() {
try { return JSON.parse(fs.readFileSync(CACHE_FILE, "utf8")); } catch (e) { return {}; }
}
function saveCache() { writeJson(CACHE_FILE, cache); }

function loadGroupIndex() {
try { return JSON.parse(fs.readFileSync(GROUP_INDEX_FILE, "utf8")); } catch (e) { return {}; }
}

function flushGroupIndex() {
if (!indexDirty) return;
const keys = Object.keys(groupIndex);
if (keys.length > 300) {
keys.sort(function (a, b) { return (groupIndex[b].lastSeen || 0) - (groupIndex[a].lastSeen || 0); });
keys.slice(300).forEach(function (k) { delete groupIndex[k]; });
}
writeJson(GROUP_INDEX_FILE, groupIndex);
indexDirty = false;
}

function scheduleFlush(ms) {
if (indexTimer) return;
indexTimer = setTimeout(function () { indexTimer = null; flushGroupIndex(); }, ms || 10000);
if (indexTimer.unref) indexTimer.unref();
}

function touchGroup(groupId, name) {
const k = String(groupId);
if (!k) return;
const e = groupIndex[k] || { messages: 0, firstSeen: Date.now() };
e.messages = (e.messages || 0) + 1;
e.lastSeen = Date.now();
if (name) e.name = String(name).slice(0, 40);
groupIndex[k] = e;
indexDirty = true;
scheduleFlush(10000);
}

// 每次提炼群档案后清零该群累计消息数。
// 这个计数只用于回答"上次复盘以来聊了多少"，不清零就会一直涨（GUI 里也一直在累加）。
function resetGroupCount(groupId) {
const k = String(groupId);
const e = groupIndex[k];
if (!e) return false;
e.messages = 0;
e.lastCountResetAt = Date.now();
indexDirty = true;
scheduleFlush(2000);
return true;
}

// 用 NapCat 的实时群列表标记「还在 / 已退出」。
// 只有拿到非空列表才敢判定（NapCat 掉线时读到空列表，那种情况什么都不动）。
function setGroupsLive(liveIds) {
const ids = Array.isArray(liveIds) ? liveIds : [];
if (!ids.length) return { marked: 0, revived: 0 };
const now = Date.now();
const live = new Set(ids.map(function (x) { return String(x); }));
let marked = 0, revived = 0;
for (const k of Object.keys(groupIndex)) {
const e = groupIndex[k];
if (live.has(k)) {
if (e.left) { delete e.left; e.leftAt = 0; revived++; indexDirty = true; }
} else if (!e.left) {
e.left = true; e.leftAt = now; marked++; indexDirty = true;
}
}
if (marked || revived) scheduleFlush(1000);
return { marked: marked, revived: revived };
}

function removeGroup(groupId) {
const k = String(groupId);
if (!groupIndex[k]) return false;
delete groupIndex[k];
indexDirty = true;
flushGroupIndex();
return true;
}

function knownGroups(includeLeft) {
const list = Object.keys(groupIndex)
.filter(function (k) { return includeLeft || !groupIndex[k].left; })
.map(function (k) {
const e = groupIndex[k] || {};
return { groupId: k, name: e.name || null, messages: e.messages || 0, lastSeen: e.lastSeen || 0, left: !!e.left, leftAt: e.leftAt || 0 };
});
list.sort(function (a, b) { return (b.lastSeen || 0) - (a.lastSeen || 0); });
return list;
}

function groupFirstSeen(groupId) {
const e = groupIndex[String(groupId)];
return e && e.firstSeen ? e.firstSeen : 0;
}

function setGroupName(groupId, name) {
const k = String(groupId);
if (!k || !name) return;
const e = groupIndex[k] || { messages: 0, firstSeen: Date.now() };
e.name = String(name).slice(0, 40);
groupIndex[k] = e;
indexDirty = true;
if (!indexTimer) {
indexTimer = setTimeout(function () { indexTimer = null; flushGroupIndex(); }, 3000);
if (indexTimer.unref) indexTimer.unref();
}
}

function getCachedReply(key, ttlSec) {
const e = cache[key];
if (!e) return null;
if (Number(ttlSec) > 0 && Date.now() - e.at > Number(ttlSec) * 1000) { delete cache[key]; return null; }
return e.reply;
}

function setCachedReply(key, reply) {
cache[key] = { reply: reply, at: Date.now() };
const keys = Object.keys(cache);
if (keys.length > 500) {
keys.sort(function (a, b) { return cache[a].at - cache[b].at; });
keys.slice(0, keys.length - 500).forEach(function (k) { delete cache[k]; });
}
saveCache();
}

function loadRules() {
if (!fs.existsSync(RULES_FILE)) {
writeJson(RULES_FILE, DEFAULT_RULES);
return structuredClone(DEFAULT_RULES);
}
try {
return deepMerge(structuredClone(DEFAULT_RULES), JSON.parse(fs.readFileSync(RULES_FILE, "utf8")));
} catch (e) {
return structuredClone(DEFAULT_RULES);
}
}

function writeJson(file, obj) {
const tmp = file + ".tmp";
fs.writeFileSync(tmp, JSON.stringify(obj, null, 2), "utf8");
fs.renameSync(tmp, file);
}

function saveRules(next) {
rules = next;
writeJson(RULES_FILE, rules);
return rules;
}

function updateRules(patch) {
return saveRules(deepMerge(rules, patch));
}

function rulesFor(groupId) {
const override = (rules.groups && rules.groups[String(groupId)]) || {};
return deepMerge(deepMerge({}, rules.global), override);
}

function setGroupRules(groupId, patch) {
const gid = String(groupId);
const next = structuredClone(rules);
if (!next.groups) next.groups = {};
next.groups[gid] = deepMerge(next.groups[gid] || {}, patch);
return saveRules(next);
}

function audit(entry) {
const rec = Object.assign({ at: Date.now() }, entry);
auditRing.unshift(rec);
if (auditRing.length > MAX_AUDIT) auditRing.length = MAX_AUDIT;
try {
fs.appendFileSync(path.join(AUDIT_DIR, "audit-" + today() + ".jsonl"), JSON.stringify(rec) + "\
", "utf8");
} catch (e) {}
return rec;
}

function recentAudit(limit) {
return auditRing.slice(0, limit || 100);
}

function saveGroupProfile(groupId, obj) {
writeJson(path.join(GROUP_DIR, String(groupId) + ".json"), obj);
for (const k of Array.from(gatesCache.keys())) if (k === String(groupId) || k.indexOf(String(groupId) + "|") === 0) gatesCache.delete(k);
}

// 本群专属参数：存在群档案文件的 params 里，由「提炼群档案」自动写入。
// 只做浅合并，键与取值的安全性由 profile.js 的 clampParams 保证。
const gatesCache = new Map();          // key -> { at, merged, ver }
const GATES_TTL = 10 * 1000;

// 全局 gates / 乘数 变了就必须重算：缓存键里带上它们的指纹，
// 否则改完全局参数后最多 10 秒内本群还在用旧合并值。
function gatesVersion(globalGates, factor) {
let s = "";
try { s = JSON.stringify(globalGates || {}); } catch (e) { s = "?"; }
return (Number(factor) || 1) + "#" + s.length + "#" + s;
}

function clearGatesCache() { gatesCache.clear(); }

function gatesFor(groupId, globalGates, factor) {
const gid = String(groupId);
const ver = gatesVersion(globalGates, factor);
const key = gid + "|" + ver;
const hit = gatesCache.get(key);
if (hit && Date.now() - hit.at < GATES_TTL) return hit.merged;
// 同一群只保留最新版本，避免旧版本的缓存堆积
for (const k of Array.from(gatesCache.keys())) if (k.indexOf(gid + "|") === 0) gatesCache.delete(k);
const prof = loadGroupProfile(gid);
const base = globalGates || {};
// applyBand：白名单过滤 + 硬边界 + 相对全局值的比例带（factor 来自互动反馈），三项都在这里兜底
const flat = applyBand((prof && prof.params) || {}, base, Number(factor) || 1);
// 把 autoJoin 下的键放回嵌套层级——deepMerge 只认嵌套结构，摊平键会被当成顶层新键而失效
const extra = {};
for (const k of Object.keys(flat)) {
if (AUTOJOIN_KEYS.indexOf(k) >= 0) {
if (!extra.autoJoin) extra.autoJoin = {};
extra.autoJoin[k] = flat[k];
} else extra[k] = flat[k];
}
const merged = Object.keys(extra).length ? deepMerge(base, extra) : base;
gatesCache.set(key, { at: Date.now(), merged: merged, ver: ver });
return merged;
}

function loadGroupProfile(groupId) {
const f = path.join(GROUP_DIR, String(groupId) + ".json");
if (!fs.existsSync(f)) return null;
try { return JSON.parse(fs.readFileSync(f, "utf8")); } catch (e) { return null; }
}

return {
get rules() { return rules; },
updateRules: updateRules,
rulesFor: rulesFor,
setGroupRules: setGroupRules,
audit: audit,
recentAudit: recentAudit,
touchGroup: touchGroup,
knownGroups: knownGroups,
groupFirstSeen: groupFirstSeen,
setGroupName: setGroupName,
getCachedReply: getCachedReply,
setCachedReply: setCachedReply,
saveGroupProfile: saveGroupProfile,
loadGroupProfile: loadGroupProfile,
gatesFor: gatesFor,
resetGroupCount: resetGroupCount,
setGroupsLive: setGroupsLive,
removeGroup: removeGroup,
clearGatesCache: clearGatesCache
};
}
