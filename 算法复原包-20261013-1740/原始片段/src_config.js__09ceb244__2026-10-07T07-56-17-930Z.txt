function isPlainObject(v) {
return v !== null && typeof v === "object" && !Array.isArray(v);
}

export function deepMerge(base, patch) {
const out = isPlainObject(base) ? Object.assign({}, base) : base;
for (const key of Object.keys(patch || {})) {
const pv = patch[key];
if (isPlainObject(pv) && isPlainObject(out[key])) out[key] = deepMerge(out[key], pv);
else out[key] = pv;
}
return out;
}

// 向后兼容：feed.lowFoodRatio 是旧版「按 token 预算算剩饭」的比例，
// 现在它表示「低饭比例」（相对账号基准余额）。老配置里已经调过的值需要搬过去。
function migrateBalance(raw, base) {
const legacy = raw && raw.feed && raw.feed.lowFoodRatio;
const hasNew = raw && raw.balance && raw.balance.lowFoodRatio !== undefined;
if (legacy !== undefined && legacy !== null && !hasNew) {
base.balance.lowFoodRatio = legacy;
}

delete base.feed.lowFoodRatio;
}
// 全局人设升级：见 migratePersona()
migratePersona(raw, base);
return base;
}

// 全局人设格式升级。
//
// 人设本身是"用户会自己改"的东西，所以升级要保守：
//   · 只有「还停在上一版内置人设」的配置才自动升级（旧文本里有特征串）；
//   · 用户自己改过（文本和上一版内置默认不一致）→ 一个字都不动，并写日志提示；
//   · 升级前把旧人设备份到 config/backup/，随时能翻回来；
//   · 无论升不升级，都补上 persona.version 标记，避免下次重复判断。
const PERSONA_VERSION = 7;
const OLD_PERSONA_MARKERS = ["token", "【对群里的人】", "【你怎么说话｜贴得最紧的一条】", "【怎么说话：别背词，别演】", "【你是个什么样的群友｜回复的时候按这个来】", "【你能做什么、做不到什么｜这条要清楚，别装】"];
function personaLooksLikeOldDefault(text) {
const s = String(text || "");
if (!s) return false;
for (const m of OLD_PERSONA_MARKERS) if (s.indexOf(m) >= 0) return true;
return false;
}

function migratePersona(raw, base) {
const saved = raw && raw.persona && typeof raw.persona.global === "string" ? raw.persona.global : "";
const savedVer = Number((raw && raw.persona && raw.persona.version) || 0);
if (savedVer >= PERSONA_VERSION || !saved) { base.persona.version = PERSONA_VERSION; return false; }
if (saved === DEFAULT_PERSONA) { base.persona.version = PERSONA_VERSION; return false; }
if (!personaLooksLikeOldDefault(saved)) {
base.persona.global = saved;             // 用户自己写的 → 原样保留
base.persona.version = savedVer || 1;
try { console.log("[dsh-qq] 全局人设是你自己改过的，本次升级没有覆盖它。想换成新版内置人设，可在面板点「恢复内置默认人设」。"); } catch (e) {}
return false;
}
// 还停在旧版内置人设 → 升级（先备份）
try {
const dir = path.join(CONFIG_DIR, "backup");
ensureDir(dir);
const f = path.join(dir, "persona-v" + (savedVer || 1) + "-" + Date.now() + ".txt");
fs.writeFileSync(f, saved, "utf8");
try { console.log("[dsh-qq] 全局人设已升级到 v" + PERSONA_VERSION + "，旧版备份在 " + f); } catch (e) {}
} catch (e) {}
base.persona.global = DEFAULT_PERSONA;
base.persona.version = PERSONA_VERSION;
try { base.__personaMigrated = true; } catch (e) {}
return true;
}

export function loadConfig() {

const merged = deepMerge(structuredClone(DEFAULT_CONFIG), raw);

if (out && out.__personaMigrated) { try { delete out.__personaMigrated; saveConfig(out); } catch (e) {} }
return out;
} catch (err) {
try { fs.copyFileSync(CONFIG_FILE, CONFIG_FILE + ".broken-" + Date.now()); } catch (e) {}
return structuredClone(DEFAULT_CONFIG);
}
}
