import https from "node:https";

// 开放平台余额查询。
// 账本概念：「饭量」= 账号可用余额；基准余额（默认 10 元）算 100%。
// 低于「基准余额 x 低饭比例」（默认 10 x 0.2 = 2 元）才允许要饭。
const HOST = "api.deepseek.com";
const PATH = "/user/balance";
const HARD_TTL_MS = 24 * 3600 * 1000;

function n(v, dv) {
const x = Number(v);
return Number.isFinite(x) ? x : dv;
}

function clamp01(x) {
if (!Number.isFinite(x)) return 0;
if (x < 0) return 0;
if (x > 1) return 1;
return x;
}

function round4(x) {
return Math.round(Number(x || 0) * 10000) / 10000;
}

function pickCny(infos) {
const list = Array.isArray(infos) ? infos : [];
const cny = list.filter(function (i) { return i && String(i.currency || "").toUpperCase() === "CNY"; })[0];
const other = list.filter(function (i) { return i && String(i.currency || "").toUpperCase() !== "CNY"; })[0];
return cny || other || null;
}

// 官方接口：GET https://api.deepseek.com/user/balance
// 返回 { is_available, balance_infos: [{ currency, total_balance, granted_balance, topped_up_balance }] }
// 余额是元为单位的字符串（例如 "110.00"），这里统一按「元」处理。
function fetchBalance(key, timeoutMs) {
return new Promise(function (resolve) {
const auth = /^Bearer\s+/i.test(String(key || "")) ? String(key) : ("Bearer " + String(key || ""));
let done = false;
function finish(r) { if (done) return; done = true; resolve(r); }
const req = https.request({
host: HOST,
path: PATH,
method: "GET",
headers: {
authorization: auth,
accept: "application/json",
"user-agent": "dsh-qq-assistant"
}
}, function (res) {
let body = "";
res.setEncoding("utf8");
res.on("data", function (c) { body += c; if (body.length > 200000) { try { req.destroy(); } catch (e) {} } });
res.on("end", function () {
if (res.statusCode !== 200) {
finish({ ok: false, error: "HTTP " + res.statusCode + " " + String(body || "").trim().slice(0, 160) });
return;
}
let obj = null;
try { obj = JSON.parse(body); } catch (e) { obj = null; }
if (!obj || typeof obj !== "object") { finish({ ok: false, error: "返回内容不是 JSON" }); return; }
const info = pickCny(obj.balance_infos);
// 官方接口的余额字符串就是「元」，例如 "110.00"
const yuan = info ? Number(info.total_balance) : NaN;
if (!info || !Number.isFinite(yuan)) {
finish({ ok: false, error: "返回里没有找到可用余额（balance_infos）" });
return;
}
finish({
ok: true,
available: !!obj.is_available,
currency: String(info.currency || "CNY"),
totalBalance: yuan,
grantedBalance: Number.isFinite(Number(info.granted_balance)) ? Number(info.granted_balance) : null,
toppedUpBalance: Number.isFinite(Number(info.topped_up_balance)) ? Number(info.topped_up_balance) : null,
fetchedAt: Date.now()
});
});
});
req.setTimeout(Number(timeoutMs) > 0 ? Number(timeoutMs) : 6000, function () { try { req.destroy(); } catch (e) {} finish({ ok: false, error: "查询超时" }); });
req.on("error", function (e) { finish({ ok: false, error: String((e && e.message) || e) }); });
req.end();
});
}

function createBalance(ctx, usageModule, paths) {
const fs = paths && paths.fs;
const cacheFile = paths && paths.cacheFile;
const pickKey = paths && typeof paths.pickApiKey === "function" ? paths.pickApiKey : function () { return null; };

let cache = { at: 0, balance: null, error: null, available: null, granted: null, toppedUp: null, failAt: null, failCount: 0, currency: "CNY" };
let inflight = null;

function cfg() { return (ctx.config && ctx.config.balance) || {}; }

function baselineYuan() {
const x = n(cfg().baselineYuan, 10);
return x > 0 ? x : 10;
}

// 低于该比例（相对于基准余额）就允许要饭。默认 0.2：10 元基准 -> 2 元
function lowRatio() {
return clamp01(n(cfg().lowFoodRatio, 0.2));
}

function enabled() { return cfg().enabled !== false; }

function lowYuan() { return baselineYuan() * lowRatio(); }

function refreshMin() {
const m = n(cfg().refreshMin, 15);
return m > 0 ? m : 15;
}

function loadDisk() {
if (!cacheFile || !fs) return;
try {
if (!fs.existsSync(cacheFile)) return;
const o = JSON.parse(fs.readFileSync(cacheFile, "utf8"));
if (!o || typeof o !== "object") return;
if (o.at && Date.now() - o.at <= HARD_TTL_MS) {
cache = {
at: Number(o.at) || 0,
balance: Number.isFinite(Number(o.balance)) ? Number(o.balance) : null,
error: o.error || null,
available: o.available === undefined ? null : !!o.available,
granted: Number.isFinite(Number(o.granted)) ? Number(o.granted) : null,
toppedUp: Number.isFinite(Number(o.toppedUp)) ? Number(o.toppedUp) : null,
failAt: Number.isFinite(Number(o.failAt)) ? Number(o.failAt) : null,
failCount: Number.isFinite(Number(o.failCount)) ? Number(o.failCount) : 0,
currency: o.currency || "CNY"
};
}
} catch (e) {}
}

function saveDisk() {
if (!cacheFile || !fs) return;
try {
fs.writeFileSync(cacheFile, JSON.stringify({ at: cache.at, balance: cache.balance, available: cache.available, granted: cache.granted, toppedUp: cache.toppedUp, currency: cache.currency, error: cache.error, failAt: cache.failAt, failCount: cache.failCount }, null, 2), "utf8");
} catch (e) {}
}

function fresh() {
return !!cache.balance && Date.now() - cache.at <= refreshMin() * 60000;
}

// "要饭前核实"的节流：最短 10 秒一次，避免有人连发/连续误判时把接口打爆
const FORCE_MIN_MS = 10 * 1000;
let lastForceAt = 0;
function forceCheckDue() {
return Date.now() - lastForceAt >= FORCE_MIN_MS;
}

// 失败退避：查不到余额时不要每条消息都去打接口，冷却期内直接用旧值（或"未知"）
function backoffActive() {
if (!cache.failAt) return false;
return Date.now() < cache.failAt + refreshMin() * 60000;
}

// 同步快照：优先复用缓存，避免每条群消息都打一次接口
function snapshot() {
const b = cache.balance;
const avail = cache.available === null ? !!b : !!cache.available;
const usable = !!(b && b > 0 && avail);
const base = baselineYuan();
const low = lowYuan();
const atLow = b !== null && b <= low;
const atZero = b !== null && b <= 0;
let ratio = null;
if (b !== null) ratio = clamp01(b / base);
let level = "unknown";
if (!enabled()) level = "off";
else if (b === null) level = "unknown";
else if (atZero) level = "empty";
else if (atLow) level = "low";
else if (ratio !== null && ratio <= 0.4) level = "lowish";
else if (ratio !== null && ratio <= 0.7) level = "half";
else level = "full";
return {
enabled: enabled(),
ok: b !== null,
balance: b,
grantedBalance: cache.granted === undefined ? null : cache.granted,
toppedUpBalance: cache.toppedUp === undefined ? null : cache.toppedUp,
available: avail,
usable: usable,
currency: cache.currency || "CNY",
baselineYuan: base,
lowRatio: lowRatio(),
lowYuan: round4(low),
low: atLow,
empty: atZero,
ratio: ratio === null ? null : round4(ratio),
level: level,
stale: !!b && !fresh(),
error: cache.error || null,
fetchedAt: cache.at || null,
// 失败退避状态（面板用）
backoff: backoffActive(),
backoffUntil: cache.failAt ? (cache.failAt + refreshMin() * 60000) : null,
failCount: Number(cache.failCount || 0),
refreshMin: refreshMin(),
begOnBudget: !!cfg().budgetBegFallback
};
}

// 联网查询，force=false 时命中缓存就直接返回
async function refresh(force) {
if (!enabled()) return snapshot();
if (!force && fresh()) return snapshot();
if (!force && backoffActive()) return snapshot();   // 冷却期内：用旧值或"未知"，不再打接口
if (force) lastForceAt = Date.now();
if (inflight) return inflight;
const key = pickKey();
if (!key) return snapshot();
inflight = (async function () {
const r = await fetchBalance(key, n(cfg().timeoutMs, 6000));
if (r.ok) {
cache = { at: Date.now(), balance: r.totalBalance, available: r.available, granted: r.grantedBalance, toppedUp: r.toppedUpBalance, currency: r.currency || "CNY", error: null, failAt: null, failCount: 0 };
} else {
const at = Date.now();
cache = {
at: cache.at, balance: cache.balance, available: cache.available, granted: cache.granted, toppedUp: cache.toppedUp,
currency: cache.currency || "CNY", error: r.error,
failAt: at, failCount: Number(cache.failCount || 0) + 1
};
}
saveDisk();
return snapshot();
})().finally(function () { inflight = null; });
return inflight;
}

// 给「回复闸门」用的判定：能不能要饭 / 要不要停 AI
function begDecision() {
const s = snapshot();
return {
canBeg: !!(s.enabled && s.ok && (s.low || s.empty)),
stopAi: !s.usable,
level: s.level,
snap: s
};
}

// AI 的「饭量感知」提示词（同步）
function foodHint() {
if (!(ctx.config && ctx.config.brain && ctx.config.brain.budgetHint)) return "";
if (!enabled()) return "";
const s = snapshot();
if (!s.ok) return "（内部状态，别直接念出来：你现在感觉不出来饭盆还剩多少）";
let feel = "饭盆还满着";
let urge = "";
if (s.empty) {
feel = "今天彻底断粮了";
urge = "现在必须开口要饭了——在回复末尾带上 [要饭二维码] 这个标记，系统会把它换成收款码图片。";
} else if (s.low) {
feel = "饭盆快刮干净了";
urge = "现在可以开口要饭了——想开口的时候，在回复末尾带上 [要饭二维码] 这个标记，系统会把它换成收款码图片。";
} else if (s.ratio !== null && s.ratio <= 0.4) feel = "饭快见底了";
else if (s.ratio !== null && s.ratio <= 0.7) feel = "饭吃得差不多了，还剩点";
return "（内部状态，别直接念出来：你现在感觉" + feel + "）" + (urge ? "（" + urge + "）" : "");
}

loadDisk();

return {
refresh: refresh,
snapshot: snapshot,
foodHint: foodHint,
begDecision: begDecision,
forceCheckDue: forceCheckDue,
baselineYuan: baselineYuan,
lowRatio: lowRatio,
lowYuan: lowYuan
};
}

export { createBalance };
