import fs from "node:fs";
import { SECRETS_FILE, CONFIG_DIR, ensureDir } from "./paths.js";

export function loadSecrets() {
ensureDir(CONFIG_DIR);
if (!fs.existsSync(SECRETS_FILE)) return {};
try { return JSON.parse(fs.readFileSync(SECRETS_FILE, "utf8")); } catch (e) { return {}; }
}

export function saveSecrets(patch) {
const next = Object.assign({}, loadSecrets(), patch || {});
ensureDir(CONFIG_DIR);
const tmp = SECRETS_FILE + ".tmp";
fs.writeFileSync(tmp, JSON.stringify(next, null, 2), "utf8");
fs.renameSync(tmp, SECRETS_FILE);
return next;
}

export function maskKey(key) {
if (!key) return null;
if (key.length <= 10) return "***";
return key.slice(0, 6) + "..." + key.slice(-4);
}

export function pickApiKey() {
const s = loadSecrets();
const fromFile = s.deepseekApiKey && String(s.deepseekApiKey).trim();
const fromEnv = process.env.DEEPSEEK_API_KEY && String(process.env.DEEPSEEK_API_KEY).trim();
return fromFile || fromEnv || null;
}
