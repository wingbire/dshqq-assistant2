function sleep(ms) { return new Promise(function (r) { setTimeout(r, ms); }); }

function findBrowser(explicit) {
if (explicit && fs.existsSync(explicit)) return explicit;
const cands = [
"C:\\Program Files (x86)\\Microsoft\\Edge\\Application\\msedge.exe",
"C:\\Program Files\\Microsoft\\Edge\\Application\\msedge.exe",
"C:\\Program Files\\Google\\Chrome\\Application\\chrome.exe",
"C:\\Program Files (x86)\\Google\\Chrome\\Application\\chrome.exe"
];
for (const c of cands) if (fs.existsSync(c)) return c;
return null;
}

// 找到含指定文字、且文本最短（最具体）的可点元素；返回 null 表示还没渲染出来
function clickByTextJs(want) {
return "(()=>{const w=" + JSON.stringify(want) + ";" +
"const all=[...document.querySelectorAll('*')].filter(e=>{const t=(e.textContent||'').trim();" +
"return t===w||t==='¥'+w||t==='￥'+w||t===w+'元'||t.includes(w);})" +
".filter(e=>{const r=e.getBoundingClientRect();return r.width>4&&r.height>4;});" +
"if(!all.length)return null;" +
"all.sort((a,b)=>(a.textContent||'').length-(b.textContent||'').length);" +
"const el=all[0];el.click();" +
"return{hit:(el.textContent||'').trim().slice(0,16),tag:el.tagName};})()";
}

// 支付方式那两行（支付宝/微信支付）读不到文本（伪元素渲染），只能靠结构定位：
// 找到文本为「支付方式」的容器，取里面按 y 排序的第 index 行。0=第一行(支付宝)，1=第二行(微信)

return "(()=>{const all=[...document.querySelectorAll('*')];" +
"const box=all.find(e=>(e.textContent||'').trim()==='支付方式'&&e.getBoundingClientRect().height>80);" +
"if(!box)return null;" +
"const rows=[...box.querySelectorAll('*')].map(e=>({e:e,r:e.getBoundingClientRect()}))" +
".filter(o=>o.r.width>300&&o.r.height>=32&&o.r.height<=80).sort((a,b)=>a.r.y-b.r.y);" +
"const uniq=[];for(const o of rows){const s=uniq.find(u=>Math.abs(u.r.y-o.r.y)<6);" +
"if(!s)uniq.push(o);else if(o.r.width<s.r.width)uniq[uniq.indexOf(s)]=o;}" +
"const pick=uniq[" + index + "];if(!pick)return null;pick.e.click();" +
"return{y:Math.round(pick.r.y),w:Math.round(pick.r.width),h:Math.round(pick.r.height),tag:pick.e.tagName,rows:uniq.length};})()";

function findQrJs(selector) {
if (selector) {
return "(()=>{const e=document.querySelector(" + JSON.stringify(selector) + ");if(!e)return null;const r=e.getBoundingClientRect();return{x:r.x,y:r.y,width:r.width,height:r.height,tag:e.tagName,how:'selector'};})()";
}
// 页面上有多个 canvas，不能只挑"最大的正方形"。
// 真正的二维码一定在支付模态框里，而模态框里有「扫一扫付款」这句话 —— 以此为锚点。
return "(()=>{" +
"const sq=function(e){const r=e.getBoundingClientRect();const s=Math.min(r.width,r.height)/Math.max(r.width,r.height);" +
"return e.tagName&&(e.tagName==='IMG'||e.tagName==='CANVAS')&&r.width>=100&&r.width<=420&&s>=0.9?{e:e,r:r,a:r.width*r.height}:null;};" +
"const all=[...document.querySelectorAll('*')];" +
"const modal=all.filter(e=>(e.textContent||'').includes('扫一扫付款')).sort((a,b)=>(a.textContent||'').length-(b.textContent||'').length)[0];" +
"const scopes=modal?[modal,document]:[document];" +
"for(const sc of scopes){" +
"const c=[...sc.querySelectorAll('img,canvas')].map(sq).filter(Boolean).sort((x,y)=>y.a-x.a);" +
"if(c.length){const o=c[0];try{window.__dshQrEl=o.e;}catch(e){}return{x:o.r.x,y:o.r.y,width:o.r.width,height:o.r.height,tag:o.e.tagName,how:modal?'modal':'page',cands:c.length};}" +
"}" +
"return null;})()";

// 用 CDP 驱动浏览器：Node 自带 WebSocket，不需要任何第三方依赖
export function createRechargeCapture(ctx, dirs) {
const feedDir = dirs.feedDir;
const profileDir = dirs.profileDir;
let running = false;
let lastResult = null;
let lastFailAt = 0;

function autoCfg() {
const f = (ctx.config.feed && ctx.config.feed.auto) || {};
return {
enabled: f.enabled !== false,
url: f.url || "https://platform.deepseek.com/top_up",
amount: f.amount == null ? "2" : String(f.amount),
method: f.method || "微信",
maxAgeMin: Number(f.maxAgeMin == null ? 10 : f.maxAgeMin),
port: Number(f.port || 9333),
browser: f.browser || "",
qrSelector: f.qrSelector || "",
amountSelector: f.amountSelector || "",
methodSelector: f.methodSelector || "",
methodRowIndex: f.methodRowIndex,
settleMs: Number(f.settleMs || 2500)
};
}

// 清掉 feed 目录里之前抓的旧二维码（按需求：每次要饭都先删旧的）
function clearOldQrs() {
const removed = [];
try {
if (!fs.existsSync(feedDir)) return removed;
for (const name of fs.readdirSync(feedDir)) {
if (!/^recharge-.*\.png$/i.test(name)) continue;
try { fs.unlinkSync(path.join(feedDir, name)); removed.push(name); } catch (e) {}
}
} catch (e) {}
return removed;
}

function currentQr() {
try {
const files = fs.readdirSync(feedDir).filter(function (n) { return /^recharge-.*\.png$/i.test(n); });
if (!files.length) return null;
files.sort();
const p = path.join(feedDir, files[files.length - 1]);
const st = fs.statSync(p);
return { path: p, at: st.mtimeMs, ageMin: (Date.now() - st.mtimeMs) / 60000 };
} catch (e) { return null; }
}

function cdp(wsUrl) {
return new Promise(function (resolve, reject) {
const ws = new WebSocket(wsUrl);
let seq = 0;
const pending = new Map();
const timer = setTimeout(function () { try { ws.close(); } catch (e) {} reject(new Error("CDP 连接超时")); }, 15000);
ws.onopen = function () { clearTimeout(timer); resolve({ ws: ws, send: send, close: function () { try { ws.close(); } catch (e) {} } }); };
ws.onerror = function () { clearTimeout(timer); reject(new Error("CDP 连接失败")); };
ws.onmessage = function (ev) {
let m; try { m = JSON.parse(ev.data); } catch (e) { return; }
const p = pending.get(m.id);
if (!p) return;
pending.delete(m.id);
if (m.error) p.reject(new Error(m.error.message || "CDP 错误"));
else p.resolve(m.result);
};
function send(method, params, timeoutMs) {
return new Promise(function (res, rej) {
const id = ++seq;
pending.set(id, { resolve: res, reject: rej });
try { ws.send(JSON.stringify({ id: id, method: method, params: params || {} })); }
catch (e) { pending.delete(id); rej(e); return; }
setTimeout(function () { if (pending.has(id)) { pending.delete(id); rej(new Error("CDP 超时: " + method)); } }, timeoutMs || 30000);
});
}
});
}

async function evaluate(conn, expr, timeoutMs) {
const r = await conn.send("Runtime.evaluate", { expression: expr, returnByValue: true, awaitPromise: true }, timeoutMs || 20000);
if (r && r.exceptionDetails) throw new Error("页面脚本出错: " + JSON.stringify(r.exceptionDetails).slice(0, 200));
return r && r.result ? r.result.value : null;
}

// 轮询等到条件成立——页面是懒渲染的，固定 sleep 靠不住
async function waitFor(conn, expr, timeoutMs, label) {
const t0 = Date.now();
while (Date.now() - t0 < timeoutMs) {
const v = await evaluate(conn, expr);
if (v) return { value: v, ms: Date.now() - t0 };
await sleep(500);
}
return { value: null, ms: Date.now() - t0, timeout: true, label: label || "" };
}

async function launch(cfg, visible) {
const exe = findBrowser(cfg.browser);
if (!exe) throw new Error("找不到 Edge 或 Chrome，请在配置里指定 feed.auto.browser");
ensureDir(profileDir);
const args = [
"--remote-debugging-port=" + cfg.port,
"--user-data-dir=" + profileDir,
visible ? "--new-window" : "--headless=new",
"--no-first-run",
"--no-default-browser-check",
"--disable-session-crashed-bubble",
cfg.url
];
spawn(exe, args, { detached: true, stdio: "ignore", windowsHide: !visible }).unref();
for (let i = 0; i < 50; i++) {
try {
const r = await fetch("http://127.0.0.1:" + cfg.port + "/json/version");
if (r.ok) return exe;
} catch (e) {}
await sleep(400);
}
throw new Error("浏览器已启动但调试端口没就绪（端口 " + cfg.port + " 可能被占用）");
}

async function pageTarget(cfg) {
for (let i = 0; i < 30; i++) {
try {
const list = await (await fetch("http://127.0.0.1:" + cfg.port + "/json/list")).json();
const pages = (list || []).filter(function (t) { return t.type === "page" && t.webSocketDebuggerUrl; });
const hit = pages.find(function (t) { return (t.url || "").indexOf("deepseek") >= 0; }) || pages[0];
if (hit) return hit;
} catch (e) {}
await sleep(400);
}
throw new Error("找不到可用的浏览器标签页");
}

function psRun(cmd) {
return new Promise(function (resolve) {
import("node:child_process").then(function (cp) {
const ps = "C:\\Windows\\System32\\WindowsPowerShell\\v1.0\\powershell.exe";
cp.execFile(ps, ["-NoProfile", "-Command", cmd], { windowsHide: true, timeout: 15000 }, function (err, stdout) { resolve(String(stdout || "").trim()); });
}).catch(function () { resolve(""); });
});
}

// 这个 profile 是不是已经有浏览器在用了（比如你正开着登录窗口）
async function profileBusy() {
const esc = profileDir.replace(/'/g, "''");
const cmd = "Get-CimInstance Win32_Process -Filter \"Name='msedge.exe' OR Name='chrome.exe'\" | Where-Object { $_.CommandLine -like '*" + esc + "*' } | Measure-Object | Select-Object -ExpandProperty Count";
const out = await psRun(cmd);
const n = parseInt(out, 10);
return isNaN(n) ? 0 : n;
}

// 只关我们自己这个 profile 的浏览器；按进程名过滤，绝不误伤别的进程
async function shutdown(cfg) {
const esc = profileDir.replace(/'/g, "''");
await psRun("Get-CimInstance Win32_Process -Filter \"Name='msedge.exe' OR Name='chrome.exe'\" | Where-Object { $_.CommandLine -like '*" + esc + "*' } | ForEach-Object { Stop-Process -Id $_.ProcessId -Force -ErrorAction SilentlyContinue }");
}
