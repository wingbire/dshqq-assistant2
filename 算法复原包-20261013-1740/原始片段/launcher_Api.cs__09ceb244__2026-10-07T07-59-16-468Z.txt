static class Api
{
public static string Base = "http://127.0.0.1:7788";
static JavaScriptSerializer ser = new JavaScriptSerializer();
public static int TimeoutMs = 20000;

public static Dictionary<string, object> Get(string path)
{
return Get(path, TimeoutMs);
}

public static Dictionary<string, object> Get(string path, int timeoutMs)
{
var req = (HttpWebRequest)WebRequest.Create(Base + path);
req.Method = "GET";
req.Timeout = timeoutMs;
req.ReadWriteTimeout = timeoutMs;
req.Accept = "application/json";
using (var resp = (HttpWebResponse)req.GetResponse())
using (var r = new System.IO.StreamReader(resp.GetResponseStream(), Encoding.UTF8))
return ser.Deserialize<Dictionary<string, object>>(r.ReadToEnd());
}

public static Dictionary<string, object> Send(string method, string path, object body)
{
var req = (HttpWebRequest)WebRequest.Create(Base + path);
req.Method = method;
req.ContentType = "application/json; charset=utf-8";
req.Timeout = TimeoutMs;
req.ReadWriteTimeout = TimeoutMs;
if (body != null)
{
byte[] bytes = Encoding.UTF8.GetBytes(ser.Serialize(body));
req.ContentLength = bytes.Length;
using (var s = req.GetRequestStream()) s.Write(bytes, 0, bytes.Length);
}
using (var resp = (HttpWebResponse)req.GetResponse())
using (var r = new System.IO.StreamReader(resp.GetResponseStream(), Encoding.UTF8))
return ser.Deserialize<Dictionary<string, object>>(r.ReadToEnd());
}

public static double Dbl(string s, double dv)
{
double d;
if (double.TryParse(s, out d)) return d;
if (double.TryParse(s, System.Globalization.NumberStyles.Float, System.Globalization.CultureInfo.InvariantCulture, out d)) return d;
return dv;
}

public static object GetPath(object root, string path)
{
object cur = root;
foreach (string k in path.Split('.'))
{
var d = cur as Dictionary<string, object>;
if (d == null || !d.ContainsKey(k)) return null;
cur = d[k];
}
return cur;
}

public static void SetPath(Dictionary<string, object> root, string path, object value)
{
string[] ks = path.Split('.');
Dictionary<string, object> cur = root;
for (int i = 0; i < ks.Length - 1; i++)
{
if (!cur.ContainsKey(ks[i]) || !(cur[ks[i]] is Dictionary<string, object>))
cur[ks[i]] = new Dictionary<string, object>();
cur = (Dictionary<string, object>)cur[ks[i]];
}
cur[ks[ks.Length - 1]] = value;
}

public static List<string> LinesToStringList(string text)
{
var list = new List<string>();
foreach (string raw in (text ?? "").Replace("\
\
", "\
").Split('\
'))
{
string s = raw.Trim();
if (s.Length > 0) list.Add(s);
}
return list;
}

public static string ListToLines(object v)
{
var arr = v as IEnumerable;
if (arr == null || v is string) return v == null ? "" : v.ToString();
var parts = new List<string>();
foreach (object o in arr) parts.Add(o == null ? "" : o.ToString());
return string.Join("\
", parts.ToArray());
}
