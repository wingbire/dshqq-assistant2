138| function isPlainObject(v) {
139|   return v !== null && typeof v === "object" && !Array.isArray(v);
140| }

142| export function deepMerge(base, patch) {
143|   const out = isPlainObject(base) ? Object.assign({}, base) : base;
144|   for (const key of Object.keys(patch || {})) {
145|     const pv = patch[key];
146|     if (isPlainObject(pv) && isPlainObject(out[key])) out[key] = deepMerge(out[key], pv);
147|     else out[key] = pv;
148|   }
149|   return out;
150| }
