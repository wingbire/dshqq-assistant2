// 判定中枢（新架构的核心前置面之一）：**事件 → 判定 → 分派到功能模块**。
//
// 按你批准的分层：
//   · **机制进前置**：这条链路本身（本文件）+ 判定纯函数（gates/gatescore/utility）
//   · **策略进功能模块**：六因子权重、门槛线、话风臂、按群参数 —— 它们可被登记、替换、学到
// 判据：删掉本文件，程序起不来（没人分派）→ 属前置；删掉权重，程序照样起 → 属功能。
//
// 【本步只做机制】判定本身**复用**旧实现（gates.decide），一个字都不重写；
// 所以这一步不改任何旧行为，基线指纹必须为 0。
import { get, list, 缺口, 状态 } from "./registry.js";

// 事件 → 该跑哪些功能模块。intent 是**归一化**过的（谁解析的报文谁负责填），
// 所以本函数是纯函数、可离线测，不碰网络也不碰旧代码。
export function route(intent) {
var it = intent || {};
var ids = [];
if (it.hasVoice) ids.push("voice-in");
if (it.hasImage) ids.push("vision-in");
if (it.hasLink) ids.push("refvideo-read");
if (it.wantsMeme) ids.push("meme-send");
if (it.replyWithVoice) ids.push("voice-out");
if (it.wantProfile) ids.push("group-profile");
// 只分派**登记过**的；没登记的不许瞎调（协议：超出能力要报缺口）
return ids.filter(function (id) { return !!get(id); });
}

// 经登记表校验后再调用（先校验后执行）。
// 【本步】所有功能模块状态都是「缺件」（还没包装唯一入口）→ 一律返回缺口，
// **绝不假装成功**、也不 404。Step 2 把包装层做出来后，这里就能真调起来。
export function invoke(id, payload, deps) {
var m = get(id);
if (!m) return { ok: false, 缺口: 缺口(id) };
if (m.状态 !== 状态.可用) return { ok: false, 缺口: 缺口(id) };
// 状态=可用 时用调用方注入的实现（Step 2 的包装层从这里接上；本步还没接，所以会走缺口上报）
var impl = (deps && typeof deps[id] === "function") ? deps[id] : null;
if (!impl) return { ok: false, 缺口: 缺口(id) };
try { return { ok: true, 结果: impl(payload) }; } catch (e) { return { ok: false, error: String((e && e.message) || e) }; }
}

// 清单（协议 ⑦三 接入层第一件：agent 读一遍就知道有哪些能力、怎么调）
export function describe() {
var 可 = list().filter(function (m) { return m.状态 === 状态.可用; }).map(function (m) { return m.编号; });
var 缺 = list().filter(function (m) { return m.状态 !== 状态.可用; }).map(function (m) { return m.编号; });
return { 可调用: 可, 缺件: 缺 };
}
