// 隐藏 NapCat 拉起的 QQ 控制台窗口（NapCat 启动器本身是隐藏的，但 QQ.exe 会自己开一个）
static void SetNapcatConsole(bool show)
{
try
{
EnumWindows(delegate(IntPtr h, IntPtr l)
{
if (!show && !IsWindowVisible(h)) return true;
var cls = new StringBuilder(256);
GetClassName(h, cls, 256);
if (cls.ToString() != "ConsoleWindowClass") return true;
var title = new StringBuilder(512);
GetWindowText(h, title, 512);
if (title.ToString().IndexOf("NapCat", StringComparison.OrdinalIgnoreCase) >= 0)
{
ShowWindow(h, show ? 5 : 0); // SW_SHOW / SW_HIDE
}
return true;
}, IntPtr.Zero);
}
catch (Exception) { }
}

// (3) exe 自带 NapCat：一键修复 / 重装注入
static void RepairNapcat()
{
var asm = Assembly.GetExecutingAssembly();
if (asm.GetManifestResourceStream("napcat.zip") == null)
{
MessageBox.Show("这个版本没有内置 NapCat 安装包。", "DSH-QQ 群小助手");
return;
}
string launcherPath = "";
try
{
var ncr = Api.Get("/api/napcat", 3000);
var nc = ncr.ContainsKey("napcat") ? (Dictionary<string, object>)ncr["napcat"] : null;
if (nc != null) launcherPath = Api.Str(nc["launcherPath"]);
}
catch (Exception) { }
if (string.IsNullOrEmpty(launcherPath))
{
MessageBox.Show("还没配置 NapCat 启动程序路径，请先在窗口「连接」页填写并保存。", "DSH-QQ 群小助手");
return;
}
string shellRoot = Path.GetDirectoryName(launcherPath);
string appTarget = null;
try
{
string verRoot = Path.Combine(shellRoot, "versions");
if (Directory.Exists(verRoot))
{
foreach (string v in Directory.GetDirectories(verRoot))
{
string cand = Path.Combine(v, "resources", "app");
if (Directory.Exists(cand)) { appTarget = cand; break; }
}
}
}
catch (Exception) { }
if (appTarget == null)
{
MessageBox.Show("没有在\
" + shellRoot + "\
下面找到 QQ 的 resources\\app 目录，无法修复。", "DSH-QQ 群小助手");
return;
}
if (MessageBox.Show("即将把内置的 NapCat 重新注入到：\
" + appTarget + "\
napcat\
\
请先确认 QQ / NapCat 没有在运行。继续吗？", "DSH-QQ 群小助手", MessageBoxButtons.OKCancel, MessageBoxIcon.Warning) != DialogResult.OK) return;

try
{
string napcatDir = Path.Combine(appTarget, "napcat");
string stagingDir = Path.Combine(appTarget, "napcat.new");
// 先解压到临时目录，成功后再替换，避免解压失败把可用安装搞坏
if (Directory.Exists(stagingDir)) Directory.Delete(stagingDir, true);
Directory.CreateDirectory(stagingDir);
string tmpZip = Path.Combine(dataDir, "napcat-restore.zip");
Directory.CreateDirectory(dataDir);
using (var s = asm.GetManifestResourceStream("napcat.zip"))
using (var f = File.Create(tmpZip)) s.CopyTo(f);
ZipFile.ExtractToDirectory(tmpZip, stagingDir);
try { File.Delete(tmpZip); } catch (Exception) { }
if (!File.Exists(Path.Combine(stagingDir, "napcat.mjs")))
{
try { Directory.Delete(stagingDir, true); } catch (Exception) { }
MessageBox.Show("内置的 NapCat 包解压后不完整，已回滚，原安装未改动。", "DSH-QQ 群小助手");
return;
}
string backupDir = Path.Combine(appTarget, "napcat.old");
if (Directory.Exists(backupDir)) Directory.Delete(backupDir, true);
if (Directory.Exists(napcatDir)) Directory.Move(napcatDir, backupDir);
Directory.Move(stagingDir, napcatDir);
try { if (Directory.Exists(backupDir)) Directory.Delete(backupDir, true); } catch (Exception) { }

string pkgPath = Path.Combine(appTarget, "package.json");
var ser = new JavaScriptSerializer();
var pkg = ser.Deserialize<Dictionary<string, object>>(File.ReadAllText(pkgPath, Encoding.UTF8));
pkg["main"] = "./napcat/napcat.mjs";
File.WriteAllText(pkgPath, ser.Serialize(pkg), new UTF8Encoding(false));
MessageBox.Show("NapCat 注入已修复完成（已用内置包重新写入）。\
\
现在可以双击 NapCat 启动程序，或用托盘菜单启动。", "DSH-QQ 群小助手");
}
catch (Exception ex)
{
MessageBox.Show("修复失败：\
" + ex.Message + "\
\
（如果是文件被占用，请先结束 QQ 进程再试）", "DSH-QQ 群小助手");
}
}

public static void StartService()
{
bool running = false;
try { running = child != null && !child.HasExited; } catch (Exception) { }
if (running) return;
if (!File.Exists(entry))
{
MessageBox.Show("找不到应用入口：" + entry, "DSH-QQ 群小助手");
return;
}
var psi = new ProcessStartInfo();
psi.FileName = nodeExe;
psi.Arguments = "\"" + entry + "\"";
psi.WorkingDirectory = appDir;
psi.UseShellExecute = false;
psi.CreateNoWindow = true;
psi.EnvironmentVariables["DSHQQ_DATA_DIR"] = dataDir;
psi.EnvironmentVariables["DSHQQ_NO_OPEN"] = "1";
try { child = Process.Start(psi); }
catch (Exception ex) { MessageBox.Show("启动后台服务失败：" + ex.Message, "DSH-QQ 群小助手"); }
RefreshMenu();
}

}

// 数据目录：放在程序所在的工作目录下。exe 在 dist/ 里时，取上一层（也就是工程根目录）。
static string ResolveDataDir()
{
string exeDir = AppDomain.CurrentDomain.BaseDirectory;
exeDir = exeDir.TrimEnd(Path.DirectorySeparatorChar, Path.AltDirectorySeparatorChar);
string name = Path.GetFileName(exeDir).ToLowerInvariant();
string baseDir = exeDir;
if (name == "dist" || name == "build" || name == "bin" || name == "out")
{
string parent = Path.GetDirectoryName(exeDir);
if (!string.IsNullOrEmpty(parent)) baseDir = parent;
}
return Path.Combine(baseDir, "dsh-data");
}

// 把老位置（%LOCALAPPDATA%\DSHQQ）里的“数据”搬过来；app/ 和 runtime/ 是程序本体，会自动重新解压，不搬。
static string MigrateLegacyData()
{
try
{
string oldDir = Path.Combine(Environment.GetFolderPath(Environment.SpecialFolder.LocalApplicationData), "DSHQQ");
if (!Directory.Exists(oldDir)) return null;
if (Path.GetFullPath(oldDir).TrimEnd(Path.DirectorySeparatorChar).Equals(
Path.GetFullPath(dataDir).TrimEnd(Path.DirectorySeparatorChar), StringComparison.OrdinalIgnoreCase)) return null;

string[] names = { "config", "data", "dsh-home", "persona", "logs" };
var moved = new List<string>();
Directory.CreateDirectory(dataDir);
foreach (string n in names)
{
string src = Path.Combine(oldDir, n);
string dst = Path.Combine(dataDir, n);
if (!Directory.Exists(src) || Directory.Exists(dst)) continue;
try { Directory.Move(src, dst); moved.Add(n); }
catch (Exception)
{
// 跨盘（C: -> E:）时 Move 一定失败，改复制；复制成功后删掉源，避免留两份
try
{
CopyDir(src, dst);
moved.Add(n + "(复制)");
try { Directory.Delete(src, true); } catch (Exception) { }
}
catch (Exception) { }
}
}
try
{
string oldLog = Path.Combine(oldDir, "launcher.log");
string newLog = Path.Combine(dataDir, "launcher.log");
if (File.Exists(oldLog) && !File.Exists(newLog)) File.Move(oldLog, newLog);
}
catch (Exception) { }

if (moved.Count == 0) return null;
return "数据已迁移到 " + dataDir + " （" + string.Join("、", moved.ToArray()) + "）";
}
catch (Exception) { return null; }
}

static void CopyDir(string src, string dst)
{
Directory.CreateDirectory(dst);
foreach (string f in Directory.GetFiles(src)) File.Copy(f, Path.Combine(dst, Path.GetFileName(f)), true);
foreach (string d in Directory.GetDirectories(src)) CopyDir(d, Path.Combine(dst, Path.GetFileName(d)));
}

static void EnsurePayload()
{
var asm = Assembly.GetExecutingAssembly();
string want = EmbeddedVersion();
string stampFile = Path.Combine(appDir, ".version");
string have = File.Exists(stampFile) ? File.ReadAllText(stampFile).Trim() : "";
bool needApp = have != want || !File.Exists(Path.Combine(appDir, "src", "main.js"));
bool needNode = !File.Exists(nodeExe);

Directory.CreateDirectory(dataDir);

if (needNode)
{
Directory.CreateDirectory(Path.GetDirectoryName(nodeExe));
using (var s = asm.GetManifestResourceStream("node.exe"))
{
if (s == null) throw new Exception("内嵌 node.exe 缺失");
using (var f = File.Create(nodeExe)) s.CopyTo(f);
}
}

if (needApp)
{
string zip = Path.Combine(dataDir, "payload.zip");
using (var s = asm.GetManifestResourceStream("payload.zip"))
{
if (s == null) throw new Exception("内嵌 payload.zip 缺失");
using (var f = File.Create(zip)) s.CopyTo(f);
}
if (Directory.Exists(appDir)) Directory.Delete(appDir, true);
Directory.CreateDirectory(appDir);
ZipFile.ExtractToDirectory(zip, appDir);
try { File.Delete(zip); } catch (Exception) { }
File.WriteAllText(stampFile, want);
}
}
