// 触发式知识注入（借鉴 SillyTavern World Info / lorebook 的做法）：
//
// 现在【经验教训】是每轮全量塞进提示词。条目一多，固定开销就一直涨。
// 这里给每条经验配一个可选触发条件（"命中就注入，没命中就不占上下文"），
// 默认（没有 keys 的条目）保持"永远注入"，所以对现有数据完全向后兼容。
//
// 复盘时可以让模型按格式产出：
//   别编游戏数据 [keys: 数值,伤害,多少,几层]
// 解析后存成 { text, keys: [...] }；注入时按当前这轮消息判断是否命中。

const MAX_KEYS = 6;
const MAX_KEY_LEN = 12;

// 简易分词：英文/数字按词，中文按 2-字滑窗（够用的近似召回，不引第三方分词）
export function tokensOf(text) {
const t = String(text || "").toLowerCase();
const out = [];
const latin = t.match(/[a-z0-9_]+/g) || [];
for (const w of latin) out.push(w);
const han = t.replace(/[^\u4e00-\u9fa5]/g, "");
for (let i = 0; i + 2 <= han.length; i++) out.push(han.slice(i, i + 2));
if (han.length === 1) out.push(han);
return out;
}

export function normalizeKeys(keys) {
const list = Array.isArray(keys) ? keys : [];
const out = [];
for (const k of list) {
let s = String(k || "").trim().replace(/^[\["']+|[\]"',，]+$/g, "");
if (!s) continue;
if (s.length > MAX_KEY_LEN) s = s.slice(0, MAX_KEY_LEN);
if (out.indexOf(s) < 0) out.push(s);
if (out.length >= MAX_KEYS) break;
}
return out;
}

// 一条经验是否需要注入
export function matches(entry, text) {
const t = String(text || "");
const keys = (entry && entry.keys) || [];
if (!keys.length) return true;                 // 没配 keys → 常驻
const toks = tokensOf(t);
for (const k of keys) {
const kk = String(k || "").toLowerCase();
if (!kk) continue;
if (kk.length >= 2 && t.toLowerCase().indexOf(kk) >= 0) return true;   // 直接子串命中
if (toks.indexOf(kk) >= 0) return true;                              // 分词命中
}
return false;
}

// 从一组经验里挑出这轮要注入的（顺序保持原样）
export function pickFor(text, entries) {
const list = Array.isArray(entries) ? entries : [];
return list.filter(function (e) { return matches(e, text); });
}

// 复盘输出的一行 -> { text, keys }
// 支持两种写法：
//   别编游戏数据 [keys: 数值,伤害,多少]
//   别报数字 [keys: 数值] [key: 层数]
export function parseLessonLine(raw) {
let s = String(raw || "").trim();
if (!s) return null;
s = s.replace(/^[-*·•]\s*/, "").replace(/^\d+[.、)]\s*/, "").trim();
const keys = [];
s = s.replace(/[\[【]\s*keys?\s*[:：]\s*([^\]】]+)[\]】]/gi, function (_m, g1) {
for (const k of String(g1).split(/[,，、|/]+/)) keys.push(k);
return "";
});
s = s.replace(/\s{2,}/g, " ").trim();
if (!s) return null;
return { text: s, keys: normalizeKeys(keys) };
}

export function parseLessonLines(list) {
const out = [];
for (const raw of (Array.isArray(list) ? list : [])) {
const e = parseLessonLine(raw);
if (e) out.push(e);
}
return out;
}

// 统一成 { text, keys } 形态（兼容历史上存的纯字符串）
export function normalizeEntries(list) {
const out = [];
for (const raw of (Array.isArray(list) ? list : [])) {
if (!raw) continue;
if (typeof raw === "string") { const e = parseLessonLine(raw); if (e) out.push(e); }
else if (raw.text) out.push({ text: String(raw.text), keys: normalizeKeys(raw.keys) });
}
return out;
}

export function formatForPrompt(entries) {
return entries.map(function (e) {
return e.keys && e.keys.length ? ("- " + e.text + "（触发词：" + e.keys.join("、") + "）") : ("- " + e.text);
});
}
