import { EventEmitter } from "node:events";

export class OneBotClient extends EventEmitter {
constructor(options) {
super();
const o = options || {};
this.options = {
url: o.url || "ws://127.0.0.1:3001",
token: o.token || "",
reconnectMs: Number(o.reconnectMs) || 3000
};
this.ws = null;
this.connected = false;
this.stopped = true;
this.pending = new Map();
this.echoSeq = 0;
this.lastError = null;
this.selfId = null;
this.connectedAt = null;
this.reconnectTimer = null;
this.backoff = this.options.reconnectMs;
}

buildUrl() {
const u = new URL(this.options.url);
if (this.options.token) u.searchParams.set("access_token", this.options.token);
return u.toString();
}

start() {
this.stopped = false;
this.backoff = this.options.reconnectMs;
this.connect();
}

stop() {
this.stopped = true;
if (this.reconnectTimer) { clearTimeout(this.reconnectTimer); this.reconnectTimer = null; }
const ws = this.ws;
this.ws = null;
if (ws) { try { ws.close(); } catch (e) {} }
this.failPending(new Error("连接已停止"));
this.setConnected(false);
}

reconfigure(next) {
const changed = next.url !== this.options.url || (next.token || "") !== this.options.token;
this.options = { url: next.url, token: next.token || "", reconnectMs: Number(next.reconnectMs) || 3000 };
if (changed && !this.stopped) { this.stop(); this.start(); }
return changed;
}

setConnected(v, err) {
const changed = this.connected !== v;
this.connected = v;
if (err !== undefined) this.lastError = err;
if (changed) this.emit("state", this.status());
}

status() {
return {
connected: this.connected,
url: this.options.url,
selfId: this.selfId,
connectedAt: this.connectedAt,
lastError: this.lastError
};
}

connect() {
if (this.stopped) return;
let ws;
try {
const url = this.buildUrl();
ws = this.options.token
? new WebSocket(url, { headers: { Authorization: "Bearer " + this.options.token } })
 new WebSocket(url);
} catch (e) {
this.lastError = String((e && e.message) || e);
this.scheduleReconnect();
return;
}
this.ws = ws;
ws.addEventListener("open", () => {
this.connectedAt = Date.now();
this.backoff = this.options.reconnectMs;
this.setConnected(true, null);
this.emit("open");
});
ws.addEventListener("message", (e) => this.onMessage(e.data));
ws.addEventListener("error", () => { this.lastError = "WebSocket 连接错误"; });
ws.addEventListener("close", () => {
if (this.ws === ws) this.ws = null;
this.setConnected(false, this.lastError || "连接已关闭");
this.failPending(new Error("连接已关闭"));
this.scheduleReconnect();
});
}

scheduleReconnect() {
if (this.stopped) return;
if (this.reconnectTimer) clearTimeout(this.reconnectTimer);
const delay = Math.max(500, this.backoff);
this.reconnectTimer = setTimeout(() => { this.reconnectTimer = null; this.connect(); }, delay);
this.backoff = Math.min(this.backoff * 2, 30000);
}

onMessage(raw) {
let msg;
try { msg = JSON.parse(typeof raw === "string" ? raw : String(raw)); }
catch (e) { return; }
if (msg && msg.echo !== undefined && this.pending.has(msg.echo)) {
const p = this.pending.get(msg.echo);
this.pending.delete(msg.echo);
clearTimeout(p.timer);
if (msg.status === "ok" || msg.retcode === 0) p.resolve(msg.data);
else p.reject(new Error("OneBot 调用失败: " + (msg.message || msg.wording || msg.retcode)));
return;
}
this.emit("event", msg);
}

call(action, params, timeoutMs) {
return new Promise((resolve, reject) => {
if (!this.connected || !this.ws) { reject(new Error("未连接到 OneBot")); return; }
const echo = "e" + (++this.echoSeq);
const timer = setTimeout(() => {
this.pending.delete(echo);
reject(new Error("OneBot 调用超时: " + action));
}, timeoutMs || 10000);
this.pending.set(echo, { resolve: resolve, reject: reject, timer: timer });
try { this.ws.send(JSON.stringify({ action: action, params: params || {}, echo: echo })); }
catch (e) { this.pending.delete(echo); clearTimeout(timer); reject(e); }
});
}

failPending(err) {
this.pending.forEach((p) => { clearTimeout(p.timer); p.reject(err); });
this.pending.clear();
}
}
