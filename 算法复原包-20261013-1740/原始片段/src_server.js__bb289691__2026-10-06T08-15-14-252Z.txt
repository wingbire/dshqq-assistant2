27| function readBody(req) {
28|   return new Promise(function (resolve, reject) {
29|     let data = "";
30|     req.on("data", function (c) { data += c; if (data.length > 2e6) { req.destroy(); reject(new Error("body too large")); } });
31|     req.on("end", function () { resolve(data); });
32|     req.on("error", reject);
33|   });

40|   const f = url && url.searchParams ? url.searchParams.get("f") : null;
41|   if (f) {
42|     const safe = path.basename(String(f));
43|     if (safe && /\.(png|jpe?g|gif|webp)$/i.test(safe)) {
44|       const cand = path.join(FEED_DIR, safe);
45|       try { if (fs.existsSync(cand) && fs.statSync(cand).isFile()) src = cand; } catch (e) {}
46|     }
47|   }
48|   if (!feed.enabled || !src || src.indexOf("http") === 0) {
49|     res.writeHead(404, { "content-type": "text/plain; charset=utf-8" });
50|     return res.end("no feed image");
51|   }
52|   try {
53|     if (!fs.existsSync(src) || !fs.statSync(src).isFile()) {
54|       res.writeHead(404); return res.end("file missing");
55|     }
56|     const ext = path.extname(src).toLowerCase();
57|     res.writeHead(200, { "content-type": MIME[ext] || "image/png" });
58|     return fs.createReadStream(src).pipe(res);
59|   } catch (e) {
