// 代价感知的互动评分（cost-aware reward）。
//
// 设计参考（搜索到的相关工作）：
//  1) "Reinforcing Real-world Service Agents: Balancing Utility and Cost in Task-oriented Dialogue"
//     (ICML 2026, InteractCS-RL)：把"共情/有用"和"预算"一起放进 RL 的 reward，做多轮成本感知决策。
//     → 对应我们这里的"互动分 - λ·token代价"。
//  2) RouteLLM 一类的 LLM 路由：在"质量-成本"之间做动态取舍，按平均输出长度/单价估成本。
//     → 对应我们按"计费 token"归一化代价，而不是按条数。
//  3) "Not all turns are equally hard: Adaptive thinking budgets for efficient multi-turn reasoning"：
//     不同轮次给不同思考预算，省 token 又不掉效果。
//     → 对应我们让"发言频率/长度/是否搜索"这类旋钮由代价感知的分数来调。
//  4) 多目标优化的经典做法：标量化（加权和 / 拉格朗日罚项）+ 近端探索（UCB / 折扣）。
//     单一标量分容易过早收敛到局部最优，所以这里加了
//       · 指数折扣：更看重最近的互动（环境会变）
//       · UCB 探索加成：样本少时给一点"再试一次"的推动
//       · 帕累托前沿：同时看互动分和 token 效率，避免"又贵又没人理"还不自知
//
// 核心公式（每条互动得一个 reward）：
//   engagement  = 已有的互动分（+4 接着聊并叫外号 / +2 接着说 / -1 没人接 / -2 嫌烦 …）
//   cost        = 归一化代价 = (未命中输入 + 2×输出)  ← 与真实计费口径一致（缓存读 ¥0.02 vs 未命中 ¥1 vs 输出 ¥4）
//   reward      = engagement - λ × cost / costScale        （λ = gates.tokenCostPenalty，默认 1.0）
//   score_ema   = 指数折扣平均（alpha 默认 0.25，约等于最近 ~4 条说话）
//   score_ucb   = score_ema + β·sqrt(refN / (1+n))          （探索加成，样本多时自动收敛）
// λ 与 costScale 都可以在配置里调：λ 调大 = 更抠门，调小 = 更看重聊天效果。

export const MAX_EXAMPLES = 5;
export const TOKEN_WEIGHT = { input: 1, output: 2 };   // 计费口径的简版加权

export function defaultScoreConfig() {
return { lambda: 1.0, alpha: 0.25, beta: 0.8, refN: 20 };
}

export function clampScoreConfig(c) {
const d = defaultScoreConfig();
const n = (v, dv) => { const x = Number(v); return Number.isFinite(x) ? x : dv; };
const cl = (x, lo, hi) => Math.min(hi, Math.max(lo, x));
return {
lambda: cl(n(c && c.lambda, d.lambda), 0, 5),
alpha: cl(n(c && c.alpha, d.alpha), 0.05, 1),
beta: cl(n(c && c.beta, d.beta), 0, 3),
refN: Math.round(cl(n(c && c.refN, d.refN), 5, 200))
};
}

// 单条互动的归一化代价
export function costOf(ev) {
const input = Number((ev && ev.tokensIn) || 0);
const output = Number((ev && ev.tokensOut) || 0);
return input * TOKEN_WEIGHT.input + output * TOKEN_WEIGHT.output;
}

// 滑动窗口的成本量纲（每条互动的平均代价），用来把 token 数缩放到"和互动分同一量级"
export function costScaleOf(events, fallback) {
const list = (events || []).filter(function (e) { return e && e.closed && e.delta !== null; });
if (!list.length) return Number(fallback) > 0 ? Number(fallback) : 3000;
let sum = 0;
for (const e of list) sum += costOf(e);
const avg = sum / list.length;
return avg > 0 ? avg : Number(fallback) > 0 ? Number(fallback) : 3000;
}

// reward + 折扣 + UCB + 帕累托
export function rewardStats(events, cfg, fallbackScale) {
const c = clampScoreConfig(cfg);
const list = (events || []).filter(function (e) { return e && e.closed && e.delta !== null; });
const n = list.length;
const costScale = costScaleOf(list, fallbackScale);
const base = {
n: n,
costScale: Math.round(costScale),
lambda: c.lambda,
avgEngagement: 0,
avgCostTokens: 0,
avgCostCostScale: 0,
scoreEma: 0,
scoreUcb: 0,
efficiency: 0,          // 互动分 / 归一化代价（越大越"省着聊还能聊"）
pareto: [],
recent: []
};
if (!n) return base;

const rows = list.map(function (e) {
const cost = costOf(e);
return { e: e, engagement: Number(e.delta), cost: cost, norm: cost / costScale };
});

let sumE = 0, sumC = 0, sumNorm = 0;
for (const r of rows) { sumE += r.engagement; sumC += r.cost; sumNorm += r.norm; }
base.avgEngagement = Math.round((sumE / n) * 100) / 100;
base.avgCostTokens = Math.round(sumC / n);
base.avgCostCostScale = Math.round((sumNorm / n) * 1000) / 1000;
base.efficiency = base.avgCostTokens > 0 ? Math.round((base.avgEngagement / base.avgCostTokens) * 1e6) / 1e6 : 0;

// 指数折扣平均（时间加权的 reward）
let ema = 0, w = 0;
for (let i = 0; i < rows.length; i++) {
const r = rows[i];
const reward = r.engagement - c.lambda * r.norm;
const weight = Math.pow(1 - c.alpha, rows.length - 1 - i);
ema += weight * reward;
w += weight;
}
ema = w > 0 ? ema / w : 0;
base.scoreEma = Math.round(ema * 1000) / 1000;
base.scoreUcb = Math.round((ema + c.beta * Math.sqrt(c.refN / (1 + n))) * 1000) / 1000;

// 帕累托前沿：既要互动分高，又要代价低（对"每条互动"做非支配筛选）
const front = [];
for (const r of rows) {
let dominated = false;
for (const o of rows) {
if (o === r) continue;
const noWorse = (o.engagement >= r.engagement) && (o.cost <= r.cost);
const better = (o.engagement > r.engagement) || (o.cost < r.cost);
if (noWorse && better) { dominated = true; break; }
}
if (!dominated) front.push({ engagement: r.engagement, cost: Math.round(r.cost), reply: String((r.e.replyText || "")).slice(0, 40) });
}
base.pareto = front.slice(-6);

base.recent = rows.slice(-6).map(function (r) {
return {
delta: r.engagement,
label: r.e.label,
costTokens: Math.round(r.cost),
reward: Math.round((r.engagement - c.lambda * r.norm) * 1000) / 1000
};
});
return base;
}

// 从历史里挑"效益最好"的发言当范例（互动分高、代价低），供下一次复盘注入提示词
export function pickExamples(events, limit) {
const list = (events || []).filter(function (e) { return e && e.closed && e.delta !== null && e.delta > 0 && e.replyText; });
const scale = costScaleOf(events, 0);
const scored = list.map(function (e) {
const cost = costOf(e);
return { e: e, value: Number(e.delta) - (cost / scale) * 0.5 };
});
scored.sort(function (a, b) { return b.value - a.value; });
const out = [];
for (const s of scored) {
if (out.length >= (limit || MAX_EXAMPLES)) break;
const t = String(s.e.replyText).trim();
if (!t) continue;
out.push({ delta: Number(s.e.delta), text: t.slice(0, 40) });
}
return out;
}

// 迭代/探索控制：正分且样本多 → 敢动（1.5）；负分且样本多 → 大幅收紧（0.5）
// 注：2026-10 修正了互动打分（"中途有人插话"不再算 +2），整体均分自然下移，
// 所以这里的档位也跟着下调一档，避免把"正常群聊"误判成"很糟"。
export function explorationFactor(scoreUcb, n) {
if (!(n >= 5)) return { factor: 1, level: "unknown" };
const a = Number(scoreUcb) || 0;
if (a >= 0.8) return { factor: 1.5, level: "good" };
if (a >= 0.3) return { factor: 1.3, level: "ok" };
if (a > -0.3) return { factor: 1, level: "neutral" };
if (a > -1) return { factor: 0.7, level: "bad" };
return { factor: 0.5, level: "veryBad" };
}
