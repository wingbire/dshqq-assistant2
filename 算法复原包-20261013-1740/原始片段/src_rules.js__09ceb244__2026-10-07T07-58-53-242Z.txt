export function createRuleState() {
return {
flood: new Map(),
floodPunish: new Map(),
atAll: new Map(),
warns: new Map()
};
}

export function parseMessage(message) {
const out = { text: "", ats: [], hasAtAll: false, replyId: null, images: 0, faces: 0 };
if (typeof message === "string") { out.text = message; return out; }
if (!Array.isArray(message)) return out;
for (const seg of message) {
if (!seg || typeof seg !== "object") continue;
const d = seg.data || {};
if (seg.type === "text") out.text += d.text || "";
else if (seg.type === "at") {
if (String(d.qq) === "all") out.hasAtAll = true;
else out.ats.push(String(d.qq));
out.text += " ";
}
else if (seg.type === "reply") out.replyId = d.id === undefined ? null : d.id;
else if (seg.type === "image") { out.images++; out.text += "[图片]"; }
else if (seg.type === "face") { out.faces++; out.text += "[表情]"; }
}
return out;
}

export function looksLikeLink(text) {
if (!text) return false;
const t = text.toLowerCase();
if (t.indexOf("http://") >= 0) return true;
if (t.indexOf("https://") >= 0) return true;
if (t.indexOf("www.") >= 0) return true;
if (t.indexOf(".com") >= 0 || t.indexOf(".cn") >= 0 || t.indexOf(".net") >= 0) {
if (t.indexOf(" ") < 0 && t.indexOf(".com") > 0) return true;
}
return false;
}

function matchBanned(text, rules) {
for (const w of rules.bannedWords || []) {
if (w && text.indexOf(w) >= 0) return "违禁词:" + w;
}
for (const p of rules.bannedRegex || []) {
if (!p) continue;
try { if (new RegExp(p, "i").test(text)) return "违禁正则:" + p; } catch (e) {}
}
return null;
}

function pushAction(actions, actionName, ctxInfo) {
if (actionName === "recall") actions.push({ kind: "recall", messageId: ctxInfo.messageId });
else if (actionName === "mute") actions.push({ kind: "mute", seconds: ctxInfo.muteSeconds || 60 });
else if (actionName === "kick") actions.push({ kind: "kick" });
else if (actionName === "warn") actions.push({ kind: "warn" });
}

function applyLadder(state, rules, groupId, userId, actions) {
const key = groupId + ":" + userId;
const count = (state.warns.get(key) || 0) + 1;
state.warns.set(key, count);
const ladder = rules.ladder || {};
const kickAfter = Number(ladder.kickAfter || 0);
const warnLimit = Number(ladder.warnLimit || 2);
if (kickAfter > 0 && count >= kickAfter) { actions.push({ kind: "kick" }); return "第 " + count + " 次违规，踢出"; }
if (count > warnLimit) { actions.push({ kind: "mute", seconds: Number(ladder.muteSeconds || 300) }); return "第 " + count + " 次违规，禁言"; }
actions.push({ kind: "warn" });
return "第 " + count + " 次违规，警告";
}

export function evaluateMessage(evt, rules, state, opts) {
const now = (opts && opts.now) || Date.now();
const result = { blocked: false, actions: [], reasons: [] };
if (!rules || rules.enabled === false) return result;

const groupId = String(evt.group_id);
const userId = String(evt.user_id);
const parsed = parseMessage(evt.message);
const text = parsed.text;
const info = { messageId: evt.message_id, muteSeconds: 0 };

if (rules.links && rules.links.enabled && looksLikeLink(text)) {
result.blocked = true;
result.reasons.push("链接");
pushAction(result.actions, rules.links.action || "recall", info);
}

if (!result.blocked) {
const hit = matchBanned(text, rules);
if (hit) {
result.blocked = true;
result.reasons.push(hit);
const detail = applyLadder(state, rules, groupId, userId, result.actions);
result.reasons.push(detail);
}
}

if (!result.blocked && rules.flood && rules.flood.enabled) {
const key = groupId + ":" + userId;
const windowMs = Number(rules.flood.seconds || 10) * 1000;
const win = (state.flood.get(key) || []).filter((t) => now - t < windowMs);
win.push(now);
state.flood.set(key, win);
const lastPunish = state.floodPunish.get(key) || 0;
if (win.length >= Number(rules.flood.count || 5) && now - lastPunish > windowMs) {
result.blocked = true;
result.reasons.push("刷屏 " + win.length + " 条/" + (windowMs / 1000) + "秒");
info.muteSeconds = Number(rules.flood.muteSeconds || 60);
pushAction(result.actions, rules.flood.action || "mute", info);
state.floodPunish.set(key, now);
state.flood.set(key, []);
}
}

if (parsed.hasAtAll && rules.atAll && rules.atAll.enabled) {
const win = (state.atAll.get(groupId) || []).filter((t) => now - t < 3600000);
win.push(now);
state.atAll.set(groupId, win);
if (win.length === Number(rules.atAll.maxPerHour || 2) + 1) {
result.blocked = true;
result.reasons.push("@全体过于频繁");
info.muteSeconds = Number(rules.atAll.muteSeconds || 300);
pushAction(result.actions, rules.atAll.action || "mute", info);
}
}

if (!result.blocked && Array.isArray(rules.autoReplies)) {
for (const r of rules.autoReplies) {
if (!r || !r.match) continue;
const matched = r.mode === "exact" ? text.trim() === r.match : text.indexOf(r.match) >= 0;
if (matched) { result.actions.push({ kind: "reply", text: r.reply || "" }); break; }
}
}

return result;
