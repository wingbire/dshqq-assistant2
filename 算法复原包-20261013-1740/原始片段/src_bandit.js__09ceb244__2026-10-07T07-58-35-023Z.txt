// 回复风格的自适应选择：按"话题桶"分配「话风臂」，用 Thompson Sampling 在线学习。
//
// 为什么这么做：
//  同一句群聊消息，可以短促接梗（便宜），也可以详细解释（贵）。原来整群只有一套参数，
//  学不出"哪个场合该短、哪个场合该详"。这里把它建模成**上下文多臂老虎机**：
//    · 臂 arm = 话风（terse 极短 / standard 常规 / detailed 详细讲解）——每臂的生成成本不同
//    · 上下文 bucket = 这轮消息的类型（提问 / 梗与吐槽 / 游戏机制 / 日常闲聊 / 其他）
//    · 收益 = 代价感知 reward（= 互动分 − λ×归一化token代价，见 score.js）
//  每次回复只在被选中的那个臂上观测到结果（反事实问题），所以我们用 **Thompson Sampling**
//  而不是贪心：按每个 (桶,臂) 的后验分布采样再取最大，天然在"利用已知好臂"和"试别的臂"之间平衡。
//
// 实现：按 **Normal-Inverse-Gamma 共轭 + Thompson Sampling**（Gaussian bandit 的标准做法）
//   · 未知均值 μ 与方差 σ²，先验 μ|σ² ~ N(μ0, σ²/κ0)，σ² ~ Inv-Gamma(a0,b0)
//   · 后验采样：σ² ~ Inv-Gamma(a_n, b_n) → μ ~ N(μ_n, σ²/κ_n)，取该臂的 μ 样本作为"这次的价值"
//   · 取样本最大的臂；冷启动时先验方差大，自然多试
//  没有外部依赖：高斯采样用 Box-Muller，Gamma 采样用 Marsaglia-Tsang，可注入 rng 便于测试。

import fs from "node:fs";
import path from "node:path";
import { DATA_ROOT, ensureDir } from "./paths.js";

const FILE = path.join(DATA_ROOT, "data", "reply-bandit.json");

// ---- 臂（话风）----
export const ARMS = ["terse", "standard", "detailed"];
export const ARM_POLICY = {
terse:    { label: "极短接话", cap: 14,  hint: "只回一句短的（10 字上下），不解释、不展开" },
standard: { label: "常规闲聊", cap: 50,  hint: "正常群聊口径，一两句，十几到四十字" },
detailed: { label: "多讲两句", cap: 140, hint: "这条值得说清楚：给结论 + 一两个要点，最多三四句" }
};

// ---- 上下文桶（消息类型）----
export const BUCKETS = ["question", "banter", "game", "smalltalk", "general"];
export const BUCKET_LABEL = {
question: "提问/求帮忙",
banter: "梗、吐槽、嘲笑",
game: "游戏机制/版本/数值",
smalltalk: "日常闲聊",
general: "其他"
};

export function bucketOf(text) {
const t = String(text || "");
if (/[?？]|吗$|呢$|怎么|为什么|如何|什么|多少|哪个|帮|求|搜|查|推荐|教我|能不能|是不是|值不值/.test(t)) return "question";
if (/哈|笑死|离谱|绝了|草|绷不住|逆天|抽象|烂|丑|傻|笨|模版|机器人|智商|崩|献祭|典/.test(t)) return "banter";
if (/塔科夫|三角洲|镜牢|困牢|红神|边狱|巴士|游戏|版本|更新|数值|伤害|装备|外挂|steam|PVP|pve|赛季|任务|副本|抽卡/.test(t)) return "game";
if (/吃|外卖|睡觉|上班|加班|天气|猫|狗|动漫|番|歌|电影|手机|电脑|显卡/.test(t)) return "smalltalk";
return "general";
}

// ---- 随机数：可注入，默认 Math.random ----
function invertGammaSample(a, scale, rnd) {
// Marsaglia-Tsang for Gamma(shape=a, scale=1)，再取倒数得到 Inv-Gamma
if (a < 1) {
const u = Math.max(1e-12, rnd());
return invertGammaSample(a + 1, scale, rnd) / Math.pow(u, 1 / a);
}
const d = a - 1 / 3;
const c = 1 / Math.sqrt(9 * d);
for (let i = 0; i < 64; i++) {
const x = gaussian(rnd);
const v = 1 + c * x;
if (v <= 0) continue;
const v3 = v * v * v;
const u = Math.max(1e-12, rnd());
if (Math.log(u) < 0.5 * x * x + d - d * v3 + d * Math.log(v3)) return (d * v3) * scale;
}
return a * scale;   // 兜底
}

function gaussian(rnd) {
let u = 0, v = 0;
while (u === 0) u = rnd();
while (v === 0) v = rnd();
return Math.sqrt(-2 * Math.log(u)) * Math.cos(2 * Math.PI * v);
}

export function createBandit(opts) {
const o = opts || {};
const file = o.file || FILE;
const rnd = typeof o.rng === "function" ? o.rng : Math.random;
// 先验：μ0=0，κ0=1（弱），a0=2，b0≈σ0²（取方差 4，能容纳 -2..+4 的 reward）
const PRIOR = { mu: 0, kappa: 1, a: 2, b: 4 };
let data = load();
let dirty = false;
let timer = null;

function load() {
try {
const j = JSON.parse(fs.readFileSync(file, "utf8"));
if (j && j.buckets) return j;
} catch (e) {}
return { buckets: {}, updatedAt: 0 };
}
function flush() {
if (!dirty) return;
try {
ensureDir(path.dirname(file));
const tmp = file + ".tmp";
fs.writeFileSync(tmp, JSON.stringify(data, null, 2), "utf8");
fs.renameSync(tmp, file);
dirty = false;
} catch (e) {}
}
function schedule() {
if (timer) return;
timer = setTimeout(function () { timer = null; flush(); }, 3000);
if (timer.unref) timer.unref();
}

function cell(bucket, arm, create) {
const b = String(bucket || "general");
const a = ARMS.indexOf(arm) >= 0 ? arm : "standard";
if (!data.buckets[b]) {
if (!create) return null;
data.buckets[b] = {};
}
if (!data.buckets[b][a]) {
if (!create) return null;
data.buckets[b][a] = { n: 0, sum: 0, sumSq: 0 };
}
return data.buckets[b][a];
}

// 从 (桶,臂) 的后验里采一个 μ 样本
function sampleMu(bucket, arm) {
const c = cell(bucket, arm, false) || { n: 0, sum: 0, sumSq: 0 };
const n = c.n;
const mean = (PRIOR.kappa * PRIOR.mu + c.sum) / (PRIOR.kappa + n);
const kappa = PRIOR.kappa + n;
const a = PRIOR.a + n / 2;
let b = PRIOR.b + 0.5 * (c.sumSq + PRIOR.kappa * PRIOR.mu * PRIOR.mu - (PRIOR.kappa * PRIOR.mu + c.sum) ** 2 / (PRIOR.kappa + n));
if (!(b > 1e-9)) b = 1e-9;
const sigma2 = invertGammaSample(a, 1 / b, rnd);   // σ² ~ Inv-Gamma(a, b)
return mean + gaussian(rnd) * Math.sqrt(sigma2 / kappa);
}

function choose(textOrBucket) {
const bucket = BUCKETS.indexOf(textOrBucket) >= 0 ? textOrBucket : bucketOf(textOrBucket);
let best = "standard", bestSample = -Infinity;
const samples = {};
for (const arm of ARMS) {
const s = sampleMu(bucket, arm);
samples[arm] = Math.round(s * 1000) / 1000;
if (s > bestSample) { bestSample = s; best = arm; }
}
return { bucket: bucket, arm: best, samples: samples, policy: ARM_POLICY[best] };
}

function record(bucket, arm, reward) {
if (!Number.isFinite(Number(reward))) return null;
const c = cell(bucket, arm, true);
const r = Number(reward);
c.n += 1;
c.sum += r;
c.sumSq += r * r;
data.updatedAt = Date.now();
dirty = true;
schedule();
return c;
}

// 给复盘用的可读摘要：每个桶里哪个臂目前最好
function summary() {
const out = [];
for (const bucket of Object.keys(data.buckets)) {
const rows = [];
for (const arm of ARMS) {
const c = cell(bucket, arm, false);
if (!c || !c.n) continue;
rows.push({ arm: arm, n: c.n, mean: Math.round((c.sum / c.n) * 1000) / 1000 });
}
if (!rows.length) continue;
rows.sort(function (a, b) { return b.mean - a.mean; });
out.push({ bucket: bucket, label: BUCKET_LABEL[bucket] || bucket, best: rows[0].arm, rows: rows });
}
return out;
}

function stats() {
let total = 0;
for (const b of Object.keys(data.buckets)) for (const a of Object.keys(data.buckets[b])) total += data.buckets[b][a].n;
return { buckets: Object.keys(data.buckets).length, samples: total, updatedAt: data.updatedAt };
}

return { choose: choose, record: record, summary: summary, stats: stats, flush: flush };
}
