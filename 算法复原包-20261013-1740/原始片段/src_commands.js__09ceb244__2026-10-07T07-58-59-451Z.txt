export function parseCommand(text, prefix) {
const t = String(text || "").trim();
const p = prefix || "/";
if (t.indexOf(p) !== 0) return null;
const body = t.slice(p.length).trim();
if (!body) return null;
const parts = body.split(" ").filter((x) => x.length > 0);
if (!parts.length) return null;
return { name: parts[0], args: parts.slice(1) };
}

export function createCommands(deps) {
const ctx = deps.ctx;
const client = deps.client;
const store = deps.store;
const actions = deps.actions;
const getStats = deps.getStats || function () { return {}; };
const summarizeGroup = deps.summarizeGroup || async function () { return { ok: false, error: "未启用" }; };
const roleCache = new Map();

async function isAdmin(evt) {
const uid = String(evt.user_id);
const gid = String(evt.group_id);
const cfg = (ctx.config.commands && ctx.config.commands.admins) || [];
if (cfg.map(String).indexOf(uid) >= 0) return true;
const key = gid + ":" + uid;
const cached = roleCache.get(key);
if (cached && Date.now() - cached.at < 60000) return cached.role === "owner" || cached.role === "admin";
try {
const info = await client.call("get_group_member_info", { group_id: Number(gid), user_id: Number(uid) });
const role = (info && info.role) || "member";
roleCache.set(key, { role: role, at: Date.now() });
return role === "owner" || role === "admin";
} catch (e) {
return false;
}
}

function summary(gid) {
const r = store.rulesFor(gid);
const lines = [];
lines.push("本群规则：" + (r.enabled === false ? "已关闭" : "已启用"));
lines.push("违禁词 " + ((r.bannedWords || []).length) + " 个 / 正则 " + ((r.bannedRegex || []).length) + " 条");
lines.push("链接拦截 " + (r.links && r.links.enabled ? "开" : "关") + " · 刷屏检测 " + (r.flood && r.flood.enabled ? "开" : "关") + " · @全体限制 " + (r.atAll && r.atAll.enabled ? "开" : "关"));
lines.push("关键词回复 " + ((r.autoReplies || []).length) + " 条");
return lines.join("\
");
}

async function handle(evt, parsed, msg) {

const uid = String(evt.user_id);
const name = parsed.name;
const args = parsed.args;
const ats = msg.ats || [];

if (!(await isAdmin(evt))) {
await actions.run({ kind: "reply", text: "只有群管理员可以用这个指令。" }, { groupId: gid, userId: uid, reason: "指令权限不足" });
return true;
}

if (name === "状态") {
const s = getStats();
await actions.run({ kind: "reply", text: "机器人运行中\
已处理消息 " + (s.messages || 0) + " 条 · 已回复 " + (s.replies || 0) + " 条\
大脑档位 " + (s.tier || "-") + " · 干跑模式 " + (s.dryRun ? "开" : "关") }, { groupId: gid, userId: uid, reason: "指令:状态" });
return true;
}

if (name === "规则") {
await actions.run({ kind: "reply", text: summary(gid) }, { groupId: gid, userId: uid, reason: "指令:规则" });
return true;
}

" + p.text) : ("还没有群档案，发送 " + (ctx.config.commands.prefix || "/") + "档案 让我提炼一份。") }, { groupId: gid, userId: uid, reason: "指令:档案查看" });
return true;

" + r.text) : ("提炼失败：" + (r.error || "未知原因")) }, { groupId: gid, userId: uid, reason: "指令:档案结果" });

}

return true;
}

if (name === "词条") {
const eff = store.rulesFor(gid);
const list = (eff.autoReplies || []).slice();
const sub = args[0];
if (sub === "加") {
const match = args[1];
const reply = args.slice(2).join(" ");
if (!match || !reply) {
await actions.run({ kind: "reply", text: "用法：" + (ctx.config.commands.prefix || "/") + "词条 加 <触发词> <回复内容>" }, { groupId: gid, reason: "指令:词条用法" });
return true;
}
const idx = list.findIndex((x) => x.match === match);
if (idx >= 0) list[idx].reply = reply; else list.push({ match: match, reply: reply, mode: "contains" });
store.setGroupRules(gid, { autoReplies: list });
await actions.run({ kind: "reply", text: "已添加词条「" + match + "」" }, { groupId: gid, userId: uid, reason: "指令:词条加" });
return true;
}
if (sub === "删") {
const match = args[1];
const next = list.filter((x) => x.match !== match);
store.setGroupRules(gid, { autoReplies: next });
await actions.run({ kind: "reply", text: "已删除词条「" + match + "」" }, { groupId: gid, userId: uid, reason: "指令:词条删" });
return true;
}
const body = list.length ? list.map((x) => x.match + " → " + x.reply).join("\
") : "（还没有词条）";
await actions.run({ kind: "reply", text: "本群词条：\
" + body }, { groupId: gid, userId: uid, reason: "指令:词条列表" });
return true;
}

if (name === "禁言" || name === "踢出" || name === "警告" || name === "解禁") {
if (!ats.length) {
await actions.run({ kind: "reply", text: "请 @ 一个目标，例如：" + (ctx.config.commands.prefix || "/") + name + " @某人" }, { groupId: gid, reason: "指令缺目标" });
return true;
}
const numArg = args.filter((a) => !isNaN(Number(a)))[0];
const minutes = numArg ? Number(numArg) : 10;
for (const target of ats) {
if (name === "禁言") await actions.run({ kind: "mute", seconds: Math.max(1, minutes * 60), userId: target }, { groupId: gid, reason: "指令:禁言" });
else if (name === "解禁") await actions.run({ kind: "unmute", userId: target }, { groupId: gid, reason: "指令:解禁" });
else if (name === "踢出") await actions.run({ kind: "kick", userId: target }, { groupId: gid, reason: "指令:踢出" });
else await actions.run({ kind: "reply", text: "@" + target + " 请注意群规。" }, { groupId: gid, reason: "指令:警告" });
}
const done = name === "禁言" ? "已禁言 " + ats.length + " 人 " + minutes + " 分钟" : name === "解禁" ? "已解除禁言" : name === "踢出" ? "已踢出" : "已警告";
await actions.run({ kind: "reply", text: done }, { groupId: gid, userId: uid, reason: "指令:" + name });
return true;
}

if (name === "撤回") {
if (!msg.replyId) {
await actions.run({ kind: "reply", text: "请回复要撤回的那条消息再发送指令。" }, { groupId: gid, reason: "指令:撤回缺目标" });
return true;
}
await actions.run({ kind: "recall", messageId: msg.replyId }, { groupId: gid, userId: uid, reason: "指令:撤回" });
return true;
}

return false;
}

return { isAdmin: isAdmin, handle: handle, summary: summary };
