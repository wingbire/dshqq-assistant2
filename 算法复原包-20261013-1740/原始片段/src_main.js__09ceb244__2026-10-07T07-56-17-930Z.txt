function fileLog(msg) {
try {
fs.appendFileSync(path.join(LOG_DIR, "app.log"), new Date().toISOString() + "  " + msg + "\
", "utf8");
} catch (e) {}
}

const VERSION = (function () {
try { return JSON.parse(fs.readFileSync(path.join(APP_ROOT, "package.json"), "utf8")).version || "0.0.0"; }
catch (e) { return "0.0.0"; }
})();
ensureBaseDirs();
let config = loadConfig();
// 把"代码里的新默认值"落盘一次：否则运行时读磁盘的那份配置会缺新键
// （例如效用判定的 gates.utility / utilitySpeakAt，缺了会退回内置默认，用户改了也不生效）
try {
const need = [
["gates", "utility"], ["gates", "utilitySpeakAt"], ["gates", "utilityTerseAt"],
["gates", "utilityDetailedAt"], ["gates", "utilityLambda"]
];
let missing = false;
for (const [a, b] of need) { if (!config[a] || config[a][b] === undefined) { missing = true; break; } }
if (missing) { saveConfig(config); }
} catch (e) {}

const client = new OneBotClient({
url: config.onebot.url,
token: config.onebot.token,
reconnectMs: config.onebot.reconnectMs
});

const ctx = {
get store() { return store; },   // 供 server.js 读写本群专属参数
version: VERSION,
startedAt: Date.now(),
dataRoot: DATA_ROOT,
dshHome: DSH_HOME,
config: config,
getDshStatus: function () { return detectDsh(config.brain.dshCliPath); },
getOnebotStatus: function () { return client.status(); },
getEvents: function () { return bot.events; },
clearEvents: function () { bot.clearEvents(); },
callOnebot: function (action, params) { return client.call(action, params); },
getStats: function () { return bot.stats; },
getRules: function () { return store.rules; },
knownGroups: function (includeLeft) { return store.knownGroups(includeLeft); },
setGroupsLive: function (ids) { return store.setGroupsLive(ids); },
removeGroup: function (gid) { return store.removeGroup(gid); },
setGroupName: function (gid, name) { return store.setGroupName(gid, name); },
updateRules: function (patch) { return store.updateRules(patch); },
setGroupRules: function (gid, patch) { return store.setGroupRules(gid, patch); },
rulesFor: function (gid) { return store.rulesFor(gid); },
getAudit: function (limit) { return store.recentAudit(limit); },
getAgentStatus: function () { return agent.status(); },
getNapcatStatus: function () { return napcat.status(); },
startNapcat: function (opts) { return napcat.start(opts); },
stopNapcat: function () { return napcat.stop(); },
getUsage: function () { return { today: usage.todayTotals(), history: usage.history(15), price: usage.price }; },
getBalance: function () { return balance.snapshot(); },
refreshBalance: function (force) { return balance.refresh(!!force); },
resetUsage: function () { return usage.resetToday(); },
summarizeGroup: function (gid) { return bot.summarizeGroup(gid); },
resetSession: function (gid) { return agent.resetSession(gid); },
getSecretsInfo: function () {
const s = loadSecrets();
return { hasKey: !!pickApiKey(), masked: maskKey(s.deepseekApiKey), fromEnv: !s.deepseekApiKey && !!process.env.DEEPSEEK_API_KEY };
},
setSecrets: function (patch) { return saveSecrets(patch); },
getRechargeStatus: function () { return Object.assign({ wired: !!ctx.recharge }, recharge.status()); },
// 本地图片要变成 NapCat 能拉的 http 地址；feed 目录里的文件带 ?f= 区分
feedImageUrl: function (p) {
const port = Number(process.env.DSHQQ_PORT || config.server.port) || 7788;
const base = "http://127.0.0.1:" + port + "/feed-qr";
try {
const abs = path.resolve(String(p || ""));
const root = path.resolve(FEED_DIR);
if (abs === root || abs.indexOf(root + path.sep) === 0) {
return base + "?f=" + encodeURIComponent(path.basename(abs));
}
} catch (e) {}
return base;
},
noteFeedError: function (msg) { bot && bot.log && bot.log({ error: "要饭二维码抓取失败：" + msg }); },
previewFeed: function () { return bot.previewFeed(); },
noteFeedUsed: function (file, kind) { bot && bot.log && bot.log({ reason: "要饭二维码(" + kind + ")：" + path.basename(String(file || "")) }); },
captureRecharge: function (opts) { return recharge.capture(opts || {}); },
openRechargeLogin: function () { return recharge.openForLogin(); },
clearRechargeQrs: function () { return { ok: true, removed: recharge.clearOldQrs() }; },
updateConfig: function (patch) {
config = deepMerge(config, patch);
saveConfig(config);
ctx.config = config;
client.reconfigure({
url: config.onebot.url,
token: config.onebot.token,
reconnectMs: config.onebot.reconnectMs
});
try { agent.applyProfilePatch(); } catch (e) {}
// 全局闸门变了 → 清掉"本群专属参数"的合并缓存，新值立刻生效（否则要等 10 秒 TTL）

};

console.log("[dsh-qq] napcat   : 异常 " + String((e && e.message) || e));

function openBrowser(url) {
try {
if (process.platform === "win32") spawn("cmd", ["/c", "start", "", url], { detached: true, stdio: "ignore", windowsHide: true }).unref();
else if (process.platform === "darwin") spawn("open", [url], { detached: true, stdio: "ignore" }).unref();
else spawn("xdg-open", [url], { detached: true, stdio: "ignore" }).unref();
} catch (e) {}
}

function shutdown() {
try { client.stop(); } catch (e) {}
server.close(function () { process.exit(0); });
setTimeout(function () { process.exit(0); }, 1500).unref();
}
process.on("SIGINT", shutdown);
process.on("SIGTERM", shutdown);
