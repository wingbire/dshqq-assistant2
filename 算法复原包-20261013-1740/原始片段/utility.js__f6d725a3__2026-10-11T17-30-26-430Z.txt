179| // 【回复字数上限的最终决策】
180| //
181| // 为什么单独抽成纯函数：2026-10-10 出过一次事故 —— 你让她「去网上搜搜再回答」，
182| // 她**真的搜了 2 次**（toolCalls=2），答案却在 **76 字**处被切在逗号后、还补了个「…」。
183| // 根因是三个坑叠在一起：
184| //   ① 上限在她开口**之前**就定死了，完全不知道她之后会不会搜索；
185| //   ② longReq 那条路是 **min**，于是「要长内容」(200) 反而被「多讲两句」(90) 压住；
186| //   ③ 那段回复被归成「对话延续」，而对话延续与主动搭话共用 autoJoin.replyCap(60) ——
187| //      60 是给**主动搭话防刷屏**用的，不该管「她接你的话」。
188| // 现在把决策集中到这里，并按下面优先级取**大**（只放宽，不无故收窄），配套单测锁住。
189| //
190| // input: { auto, continuation, longReq, toolCalls, depthCap }
191| // opts:  { autoReplyCap, longReqCap }
192| export function effectiveCap(input, opts) {
193|   const i = input || {};
194|   const o = opts || {};
195|   const num = function (v, d) { const n = Number(v); return isFinite(n) && n > 0 ? n : d; };
196|   // 基准：深度档（terse 30 / standard 90 / detailed 200）
197|   let cap = num(i.depthCap, ACTION_POLICY.standard.cap);
198|   // 【主动搭话】只有「没人叫它、它自己插嘴」才用 autoReplyCap 防刷屏；
199|   // **对话延续不算主动搭话** —— 她是在接你的话，用闲聊预算把它掐在 60 字是错的。
200|   if (i.auto && !i.continuation) cap = num(o.autoReplyCap, 60);
201|   // 【长请求】取大：要长内容时不该比「多讲两句」更短
202|   if (i.longReq) cap = Math.max(cap, num(o.longReqCap, 120));
203|   // 【真的搜索过】至少给 detailed(200)：搜索结果天生是「结论 + 几点细节」，闲聊那点字装不下
204|   if (Number(i.toolCalls) > 0) cap = Math.max(cap, ACTION_POLICY.detailed.cap);
205|   // 【P1 只往宽】该群该档的自学习偏移（replyshape 给的实测结论）：可以加长，绝不允许缩短 ——
206|   // 缩回去要么你手动改配置，要么等我把"往窄调"也做成需要人工确认的流程。
207|   const off = Math.max(0, Math.min(30, Math.round(Number(i.lenOffset) || 0)));
208|   return Math.max(1, Math.round(cap + off));
209| }
