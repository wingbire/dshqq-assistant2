@echo off
setlocal
set "PORT=17890"
set "FOUND="
for /f "tokens=5" %%a in ('netstat -ano ^| findstr ":%PORT% " ^| findstr LISTENING') do (
  echo Stopping PID %%a
  taskkill /PID %%a /F >nul 2>nul
  set "FOUND=1"
)
if not defined FOUND echo No server is listening on port %PORT%.
timeout /t 1 /nobreak >nul
endlocal
