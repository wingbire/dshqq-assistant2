// 「要不要开口」的期望值判定（voice decision）。
//
// 原来这三条路各自掷骰子：
//   · 随机触发   Math.random() < randomRate
//   · 对话延续   Math.random() < continueRate
//   · 主动参与   score >= 门槛 && Math.random() < rate
// 现在换成一条统一的判据：
//
//   开口 ⟺ P(这句话是对我说) > 门槛
//   门槛 = rate(S) / (参与收益 · 回复需求 · 长度)     ← 成本项越大，门槛越高（越不愿意开口）
//
// 其中 P 就是 addressee 模块算的置信度（已用真实群聊数据训练过）；
// 收益/成本都用**群档案里已在累积的量**估计，不需要新数据、不额外调用模型：
//   · 参与收益：该群最近互动的平均分（正分占比）——越接得住，越值得开口
//   · 回复需求：demandScore（没要求回应的消息，开口多半是自嗨）
//   · 长度成本：长消息要更长的回复 → 更贵，门槛更高
//
// 调参入口不变：本群/全局的 rate 依然是"这个群愿意多爱说话"的旋钮（越大越爱说），
// 只是它从"骰子概率"变成了"按期望值折算后的开口门槛"。

// 估算"这条消息是冲我来的"概率。做不了（没有 addressee 模块/没有上下文）时返回 null →
// 调用方退回旧行为，保证功能不会因为缺依赖而失效。
export function intentProbability(judge, input) {
if (typeof judge !== "function") return null;
try {
const jr = judge(input || {});
if (!jr || typeof jr.conf !== "number") return null;
return { p: Math.max(0, Math.min(1, jr.conf)), jr: jr };
} catch (e) { return null; }
}

// 参与收益档位：0.5（这个群基本不接我的话）~ 1.6（很接）
export function engagementFactor(avgReward) {
const a = Number(avgReward);
if (!Number.isFinite(a)) return 1;
if (a >= 1.5) return 1.6;
if (a >= 0.8) return 1.3;
if (a >= 0.2) return 1.1;
if (a > -0.2) return 1;
return 0.7;
}

// 长度成本档位：短消息便宜（1.0），长消息贵（门槛抬高）
export function lengthFactor(text) {
const n = String(text || "").length;
if (n <= 12) return 1.15;
if (n <= 40) return 1;
if (n <= 120) return 0.85;
return 0.7;
}

// 意愿参数 → 基准门槛。
//
// 方向必须是"越大越爱说"（和面板上的"概率/意愿"直觉一致）：
//   rate / continueRate 越大 → 门槛越低 → 越愿意开口。
// 锚点按历史参数的经验范围标定，中间线性插值：
//   0.15 → 0.55（很谨慎，只在明显是对我说时开口）
//   0.25 → 0.35（默认）
//   0.40 → 0.22（爱插话）
//   0.60 → 0.15
//   0.95 → 0.10（几乎有话题就接）
const WANT_ANCHORS = [[0.05, 0.90], [0.15, 0.55], [0.25, 0.35], [0.40, 0.22], [0.60, 0.15], [1, 0.10]];
export function thresholdForRate(rate) {
const r = Math.max(0.01, Math.min(1, Number(rate) || 0.25));
for (let i = 0; i < WANT_ANCHORS.length - 1; i++) {
const a = WANT_ANCHORS[i], b = WANT_ANCHORS[i + 1];
if (r >= a[0] && r <= b[0]) {
const k = (r - a[0]) / (b[0] - a[0]);
return a[1] + (b[1] - a[1]) * k;
}
}
return 0.10;
}

// 开口门槛 = 意愿基准 × 需求修正 ÷ 收益与长度修正
export function voiceThreshold(rate, factor, demandLow) {
const base = thresholdForRate(rate);
const f = Math.min(2, Math.max(0.2, Number(factor) || 1));
const t = (base * (demandLow ? 1.6 : 1)) / f;
return Math.max(0.03, Math.min(0.95, t));
}

// 统一判定。确认不了"是不是在对我说"时返回 voice=false + reason=no-confidence，
// 让调用方自己决定（我们让它沿用旧的随机行为，避免突然变哑）。
export function voiceDecision(opts) {
const o = opts || {};
const p = o.p;                       // 置信度，null 表示算不出来
const rate = o.rate;
const factor = o.factor;
const demandLow = !!o.demandLow;
const threshold = voiceThreshold(rate, factor, demandLow);
if (p === null || p === undefined) return { voice: null, p: null, threshold: threshold, reason: "no-confidence" };
return { voice: p >= threshold, p: Math.round(p * 1000) / 1000, threshold: Math.round(threshold * 1000) / 1000, factor: Math.round(factor * 100) / 100 };
}"}]
