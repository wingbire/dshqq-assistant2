// 收件人（addressee）判定：判断"这句话是不是在跟小助手说"。
//
// 设计依据（多轮对话 / 多方言谈的公开研究）：
//   · Ouchi & Tsuboi, EMNLP 2016《Addressee and Response Selection for Multi-Party Conversation》
//     —— 把"对谁说"与"该怎么回"建成两个任务，靠话语结构与篇章特征推断，而不是只靠 @。
//   · Gu et al., ACL 2021《MPC-BERT》—— 关键是把对话建成"谁对谁说了什么"的三方结构；
//     自监督信号包括 reply-to 关系、同一说话人、指针一致性。
//   · Penzo et al., EMNLP 2024《Do LLMs suffer from Multi-Party Hangover?》
//     —— 提出 Addressee Recognition 与 Response Selection 两个诊断任务，并指出性能下降主要来自
//       **对话结构复杂度**（线程交错、多话题并行），所以判定要按结构分层，而不是一刀切。
//
// 落到我们这里的做法：**用本地信号算一个置信度，不额外调用模型**（多一次调用等于每轮翻倍成本）。
// 三层信号：
//   第 0 层 硬信号：回复我 / @我 / 叫我外号（置信度 1.0）
//   第 1 层 结构信号：轮次邻接（我发完几秒内有人说话）、应答型气质、我这边有未回答的提问
//   第 2 层 语义信号：新消息在语义上更接"我上一条"还是更接"插话者那条"（群内 TF-IDF 加权的字符 n-gram 余弦）
//   第 3 层 行为统计：这个人的接话倾向、最近说话人数、距我上一条的轮次数

import fs from "node:fs";

const MAX_CORPUS = 400;
const NGRAM_N = 2;

function clamp01(x) {
if (!Number.isFinite(x)) return 0;
if (x < 0) return 0;
if (x > 1) return 1;
return x;
}

// 中文按 2-gram 切，英文/数字按词切——不引第三方分词器，够用且稳定
// 「在聊我」判别：不是我的人称 + 第三人称指代 + 评价性动词。
// 这是收件人识别里最经典的难点——"talking about" 会被误判成 "talking to"。
const THIRD_PERSON_RE = /(她|它|这鱼|这机器人|这ai|这AI|人机|机器人)/;
const ABOUT_VERB_RE = /(说|让|是|明显|压根|还|又|不|没|会|能|被|把|给|问|叫|跟|和)/;
const TO_ME_RE = /(你|您|帮我|教你|问下|问一下|看看|能不能|可不可以|有空吗|在吗)/;

export function looksAboutBot(text, names) {
const t = String(text || "");
if (!t) return false;
if (TO_ME_RE.test(t)) return false;                    // 有第二人称/请求 → 是在跟我说
const list = (names || []).map(function (x) { return String(x || "").trim(); }).filter(Boolean);
const named = list.some(function (n) { return t.indexOf(n) >= 0; });
if (!THIRD_PERSON_RE.test(t) && !named) return false;   // 没有指代我的词，也不是在叫我
if (ABOUT_VERB_RE.test(t)) return true;                 // "大肥鱼明显是乱编的" / "她说的是15层"
return t.length <= 12;                                  // "人机完了" / "好吧，确实很人机"
}

export function tokensOf(text) {
const t = String(text || "").toLowerCase();
const out = [];
const latin = t.match(/[a-z0-9]+/g) || [];
for (const w of latin) out.push("w:" + w);
const han = t.replace(/[^\u4e00-\u9fa5]/g, "");
for (let i = 0; i + NGRAM_N <= han.length; i++) out.push(han.slice(i, i + NGRAM_N));
if (han.length === 1) out.push(han);
return out;
}

// 默认权重（用真实群聊数据在 build\	rain-addressee.mjs 里网格搜索后写回 data\addressee-weights.json）
export const DEFAULT_WEIGHTS = {
recency: 0.45,      // 轮次邻接
tone: 0.25,         // 应答型气质
pending: 0.2,       // 我有未被回答的提问
semantic: 0.3,      // 语义更接谁
behavior: 0.2,      // 这个人的接话倾向
twoSpeakers: 0.1,   // 当前只有两人在聊
interjectPenalty: 0.4,  // 有人插过嘴、而这句话语义上也不像接我 → 大概率在回插话的人，扣分
aboutMePenalty: 0.45,   // 这句话是在"用第三人称聊我"（她/它/这鱼/人机 + 说/让/是…）→ 不是在跟我说话
topicAboutMePenalty: 0.25, // 连着几条都在聊我，而这条既没叫我也没有第二人称 → 是在围观我
jointBoost: 0.0,        // 事件刚发生（轮次邻接高）且语义也接我 → 两个信号互相印证，额外加成
hi: 0.6,            // 置信度 ≥ hi → +2
mid: 0.35,          // ≥ mid → +1
low: 0.2            // ≥ low → 0，低于则交给调用方判罚
};

export function createAddressee(opts) {
const o = opts || {};
const now = typeof o.now === "function" ? o.now : function () { return Date.now(); };
// 权重：外部注入 > 文件 > 默认（训练脚本会把调好的权重写进 data\addressee-weights.json）
let W = Object.assign({}, DEFAULT_WEIGHTS);
if (o.weights && typeof o.weights === "object") W = Object.assign(W, o.weights);
else if (o.weightsFile) {
try {
if (fs.existsSync(o.weightsFile)) W = Object.assign(W, JSON.parse(fs.readFileSync(o.weightsFile, "utf8")));
} catch (e) {}
}
function weights() { return Object.assign({}, W); }
function setWeights(w) { if (w && typeof w === "object") W = Object.assign(W, w); }
// 每个群一份语料与行为统计
const state = new Map();   // gid -> { df: Map, docs: number, perUser: Map, corpus: string[] }

function st(gid, create) {
const k = String(gid);
let s = state.get(k);
if (!s && create) {
s = { df: new Map(), docs: 0, perUser: new Map(), corpus: [] };
state.set(k, s);
}
return s || null;
}

// 在线学习：把群里的每句话喂进来，更新 IDF 语料与"接话倾向"
function observe(gid, msg) {
const m = msg || {};
const text = String(m.text || "").trim();
if (!text) return;
const s = st(gid, true);
const toks = new Set(tokensOf(text));
for (const t of toks) s.df.set(t, (s.df.get(t) || 0) + 1);
s.docs++;
s.corpus.push(text.slice(0, 120));
if (s.corpus.length > MAX_CORPUS) s.corpus.splice(0, s.corpus.length - MAX_CORPUS);
// 衰减旧词频，让分布跟着群话题走
if (s.docs % 200 === 0) {
for (const [t, c] of Array.from(s.df.entries())) {
const nd = Math.floor(c * 0.85);
if (nd <= 0) s.df.delete(t); else s.df.set(t, nd);
}
s.docs = Math.max(20, Math.floor(s.docs * 0.85));
}
// 行为统计：这个人说话前，上一条是不是小助手（=他是不是常接小助手的话）
const uid = String(m.userId || "");
if (uid) {
const u = s.perUser.get(uid) || { msgs: 0, afterBot: 0 };
u.msgs++;
if (m.afterBot) u.afterBot++;
s.perUser.set(uid, u);
}
}

// 用群内 IDF 算两句话的加权余弦相似度
function cosine(gid, a, b) {
const s = st(gid, false);
const ta = tokensOf(a), tb = tokensOf(b);
if (!ta.length || !tb.length) return 0;
const A = new Map(), B = new Map();
for (const t of ta) A.set(t, (A.get(t) || 0) + 1);
for (const t of tb) B.set(t, (B.get(t) || 0) + 1);
const idf = function (t) {
const df = s ? (s.df.get(t) || 0) : 0;
const N = s ? Math.max(1, s.docs) : 1;
return Math.log((N + 1) / (df + 1)) + 1;
};
let dot = 0, na = 0, nb = 0;
for (const [t, c] of A) { const w = c * idf(t); na += w * w; }
for (const [t, c] of B) { const w = c * idf(t); nb += w * w; }
for (const [t, c] of A) {
const c2 = B.get(t);
if (!c2) continue;
dot += (c * idf(t)) * (c2 * idf(t));
}
if (!na || !nb) return 0;
return dot / (Math.sqrt(na) * Math.sqrt(nb));
}

// 词级 Jaccard：中文按 1 字切（补 2-gram 在对短句时的不足）
function wordJaccard(a, b) {
const A = new Set(), B = new Set();
const ta = String(a || "").toLowerCase().match(/[\u4e00-\u9fa5]|[a-z0-9]+/g) || [];
const tb = String(b || "").toLowerCase().match(/[\u4e00-\u9fa5]|[a-z0-9]+/g) || [];
for (const x of ta) A.add(x);
for (const x of tb) B.add(x);
if (!A.size || !B.size) return 0;
let inter = 0;
for (const x of A) if (B.has(x)) inter++;
return inter / (A.size + B.size - inter);
}

// 综合相似度：字面包含 > 最大(词级 Jaccard, 字符 2-gram 余弦)
function sim(a, b) {
const A = String(a || "").trim(), B = String(b || "").trim();
if (!A || !B) return 0;
if (A.length >= 2 && B.length >= 2 && (A.indexOf(B) >= 0 || B.indexOf(A) >= 0)) return 1;
return Math.max(wordJaccard(A, B), cosine(A, B));
}

// 主判定：返回置信度与依据
//   input = { text, userId, mentionsBot, replyToBot, lastBotText, lastBotAt,
//             interjections:[{userId,text,at}], myUnansweredQuestion }
function judge(gid, input) {
const m = input || {};
const text = String(m.text || "").trim();
const parts = [];
// ---- 第 0 层：硬信号 ----
if (m.replyToBot) { parts.push({ k: "回复了我", w: 1, v: 1 }); }
if (m.mentionsBot) { parts.push({ k: "叫了我", w: 1, v: 1 }); }
if (parts.length) {
return { conf: 1, level: "certain", signals: parts, kind: "hard" };
}
// ---- 第 1 层：结构信号 ----
let struct = 0;
const dt = (m.lastBotAt && now) ? Math.max(0, now() - m.lastBotAt) : null;
let recency = 0;
if (dt !== null) {
if (dt <= 4000) recency = 1;
else if (dt <= 15000) recency = 0.75;
else if (dt <= 40000) recency = 0.5;
else if (dt <= 90000) recency = 0.25;
else recency = 0.1;
}
if (recency) { parts.push({ k: "轮次邻接", w: W.recency, v: recency }); struct += W.recency * recency; }
// 应答型气质：像在回应（疑问/附和/反驳/第二人称指代）→ 加分；像开启新话题 → 减分
// 应答型气质：疑问 / 第二人称与指代 / 附和 / 反驳 / 追问
// 注意：这里不再把「她/它/这鱼/这机器人」算成"像在回应我"——它们是第三人称，更可能是在聊我
// 应答形状：问句（含句末语气词）、第二人称、请求动作、致谢/追问
const responsive = /[?？]|你|您|咋|怎么|为什么|什么|多少|几|吗|呢|吧|嘛|有没有|真的假的|确实|谢谢|帮|搜|查|说|讲|推荐|值不值|教教|行不行|能不能/.test(text);
const initiation = /^(话说|对了|顺带|突然想到|问一下|谁知道)|\.{2,}$/.test(text);
let tone = 0;
if (responsive && text.length <= 30) tone = 1;
else if (responsive) tone = 0.6;
if (initiation) tone -= 0.8;
tone = Math.max(-1, Math.min(1, tone));
if (tone) { parts.push({ k: "应答气质", w: W.tone, v: tone }); struct += W.tone * tone; }
// 我提了个问题还没人回答 → 下一条更可能是在回我
const pending = m.myUnansweredQuestion ? 1 : 0;
if (pending) { parts.push({ k: "我有个没被回答的问题", w: W.pending, v: 1 }); struct += W.pending; }
// ---- 第 1.5 层：是在跟我聊，还是在聊我 ----
// (a) 明确回复了别人 → 基本不可能是对我说的（最可靠的一条）
if (m.replyToOther) {
return { conf: 0.05, level: "no", signals: parts.concat([{ k: "回复的是别人", w: -1, v: 1 }]), kind: "inferred", sem: null, aboutMe: true };
}
let aboutMe = 0;
// (b) 这一句在用第三人称聊我
if (!m.mentionsBot && !m.replyToBot && looksAboutBot(text, m.botNames)) {
aboutMe += W.aboutMePenalty;
parts.push({ k: "在聊我（第三人称指代）", w: -W.aboutMePenalty, v: 1 });
} else if (!m.mentionsBot && !m.replyToBot && Number(m.topicAboutMe || 0) >= 2 && !TO_ME_RE.test(text)) {
// (c) 连着几条都在聊我，而这条既没叫我也没有第二人称 → 是在围观我
aboutMe += W.topicAboutMePenalty;
parts.push({ k: "连着几条都在聊我", w: -W.topicAboutMePenalty, v: 1 });
}

// ---- 第 2 层：语义信号 ----
let sem = 0;
let semActive = false;
const botText = String(m.lastBotText || "");
if (botText && text) {
const sb = sim(text, botText);
let so = 0, otherText = "";
for (const it of (m.interjections || [])) { otherText += " " + String(it.text || ""); }
if (otherText.trim()) so = sim(text, otherText);
// 两边都没相似度 → 语义层没有信息量，不产生信号（否则会用 0 把置信度无故压低）
if (sb > 0 || so > 0) {
// 【重要】语义相似只有在"这句话本身有回应形状"时才算满。
// 否则会出现"把我的话复述给别人听"被当成"在接我"——群里讨论它时最常见的误判。
const semScale = (tone > 0) ? 1 : 0.25;
sem = (sb - so) * semScale;
semActive = true;
parts.push({ k: "语义更接" + (sem > 0 ? "我" : sem < 0 ? "别人" : "两边都不像") + (semScale < 1 ? "(无回应形状·折减)" : ""), w: W.semantic, v: Math.max(-1, Math.min(1, sem)) });
}
}
// 有人插过嘴、而这句话在语义上也不像接我 → 大概率是在回插话的那个人（文献里"线程交错"的主要误判来源）
const hadInter = Array.isArray(m.interjections) && m.interjections.length > 0;
let penalty = 0;
if (hadInter && (!semActive || sem <= 0)) {
penalty = W.interjectPenalty;
parts.push({ k: "中途有人插话且不像接我", w: -penalty, v: 1 });
}
// 事件刚发生 + 语义也接我 → 两个独立信号互相印证
let boost = 0;
if (W.jointBoost && recency >= 0.75 && semActive && sem > 0) {
boost = W.jointBoost;
parts.push({ k: "新鲜度与语义互证", w: boost, v: 1 });
}
// ---- 第 3 层：行为统计 ----
let beh = 0;
const s = st(gid, false);
const uid = String(m.userId || "");
if (s && uid) {
const u = s.perUser.get(uid);
if (u && u.msgs >= 5) {
const ratio = u.afterBot / u.msgs;           // 他说话前是小助手的比例
beh = clamp01((ratio - 0.2) / 0.5);          // 0.2 以下算 0，0.7 以上算 1
parts.push({ k: "他的接话倾向", w: W.behavior, v: beh });
}
}
const speakers = Number(m.recentSpeakers || 0);
if (speakers === 2) { parts.push({ k: "当前只有两人在聊", w: W.twoSpeakers, v: 1 }); beh = Math.min(1, beh + 1); }

const raw = struct + W.semantic * sem + W.behavior * beh - penalty - aboutMe + boost;
const conf = clamp01(raw);
const level = conf >= W.hi ? "high" : conf >= W.mid ? "mid" : conf >= W.low ? "low" : "no";
return { conf: Math.round(conf * 1000) / 1000, level: level, signals: parts, kind: "inferred", sem: semActive ? Math.round(sem * 1000) / 1000 : null, aboutMe: aboutMe > 0 };
}

// 置信度 → 互动分（连续，替代原来的二值 0/+2）
//   注意：这里只负责"是不是在跟我说话"这一层；"他到底接没接我"由 feedback 决定。
function deltaForJudge(jr) {
const c = jr && typeof jr.conf === "number" ? jr.conf : 0;
const hard = jr && jr.kind === "hard";
if (c >= W.hi) return { delta: 2, label: hard ? "明确对我说" : "顺着我的话接" };
if (c >= W.mid) return { delta: 1, label: "像是在对我说" };
if (c >= W.low) return { delta: 0, label: "不确定在对谁说" };
return { delta: null, label: "没在跟我说" };   // null = 交给调用方决定（可能 -1 / 0）
}

function stats(gid) {
const s = st(gid, false);
if (!s) return { docs: 0, users: 0 };
return { docs: s.docs, users: s.perUser.size, vocab: s.df.size };
}

return { observe: observe, judge: judge, deltaForJudge: deltaForJudge, similarity: sim, cosine: cosine, stats: stats, weights: weights, setWeights: setWeights };
}
