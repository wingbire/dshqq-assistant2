// 随机搜索 + 硬约束：误判(把别人的话当成对我说) <= 2，在此前提下最大化召回
const fs = require('fs');
const ROOT = 'E:/dshQ群小助手配置端';
async function main() {
const mod = await import('file:///E:/dshQ群小助手配置端/src/addressee.js');
const CORPUS = fs.readFileSync(ROOT + '/build/train/corpus-real.jsonl', 'utf8').trim().split(String.fromCharCode(10)).map(function (l) { return JSON.parse(l); });
const ALL = fs.readFileSync(ROOT + '/build/train/scenarios.jsonl', 'utf8').trim().split(String.fromCharCode(10)).map(function (l) { return JSON.parse(l); });
const GID = '255452871', NOW = 1800000000000, NAMES = ['大肥鱼', '鲸鲸', '蓝色大肥鱼', 'ds'];
const toBotN = ALL.filter(function (s) { return s.expect === 'toBot'; }).length;
function conf(W, s) {
const ad = mod.createAddressee({ now: function () { return NOW; }, weights: W });
for (const c of CORPUS) ad.observe(GID, { text: c.text, userId: 'corpus' });
const botAt = NOW - Number(s.gap || 0);
const inter = []; let clock = botAt + 1000;
for (const it of (s.inter || [])) { ad.observe(GID, { text: it.text, userId: it.user, afterBot: true }); inter.push({ userId: it.user, text: it.text, at: clock }); clock += 1000; }
const jr = ad.judge(GID, { text: s.msg, userId: s.user || 'u1', mentionsBot: !!s.mentionsBot, replyToBot: false,
replyToOther: !!s.replyToOther, topicAboutMe: Number(s.topicAboutMe || 0), botNames: NAMES,
lastBotText: s.bot, lastBotAt: botAt, interjections: inter,
myUnansweredQuestion: /[?？]\s*$/.test(String(s.bot || '')), recentSpeakers: Number(s.speakers || 3) });
return jr.conf;
}
function ev(W) {
let waste = 0, miss = 0;
for (const s of ALL) {
const c = conf(W, s); const said = c >= 0.35;
if (s.expect === 'toBot') { if (!said) miss++; } else { if (said) waste++; }
}
return { waste: waste, miss: miss, recall: (toBotN - miss) / toBotN };
