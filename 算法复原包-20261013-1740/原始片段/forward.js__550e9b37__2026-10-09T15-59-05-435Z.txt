10| export function sanitizeForwardId(value) {
11|   const s = String(value ?? '').trim();
12|   if (!s || s.length > 256) return '';
13|   // 仅用于 OneBot JSON 参数：允许常见 QQ 资源 id 字符，拒绝空白/控制字符。
14|   if (!/^[\w:.\-/=+@]+$/.test(s)) return '';

175| export function formatForwardResponse(data, options = {}) {
176|   const maxMessages = Math.max(1, Math.min(100, Number(options.maxMessages) || 50));
177|   const maxCharsPerMessage = Math.max(1, Math.min(2000, Number(options.maxCharsPerMessage) || 500));
178|   const raw = Array.isArray(data) ? data : (Array.isArray(data?.messages) ? data.messages : []);
179|   const total = raw.length;
180|   const truncatedByCount = total > maxMessages;
181|   let textTruncated = false;
182|   const messages = raw.slice(0, maxMessages).map((node, index) => {
183|     const fullText = nodeText(node);
184|     // 切点落在代理对中间时必须前移一个码元（判据与 md-to-plain.js 的 splitForQQ 一致）：
185|     // `slice(0, max)` 是按 UTF-16 码元切的，emoji/生僻字会被切成孤立高代理，
186|     // 那半个字符既不是合法文本，写进提示词或日志后还会渲染成乱码或替换符。
187|     let cut = maxCharsPerMessage;
188|     const cutCode = fullText.charCodeAt(cut - 1);
189|     if (cutCode >= 0xd800 && cutCode <= 0xdbff) cut -= 1;
190|     // 与 splitForQQ 相同：切点被逼到 0 时至少保留一个完整码点，
191|     // 否则 maxCharsPerMessage=1 且开头是 emoji 会让整条预览变成空串。
192|     if (cut === 0) {
193|       const first = fullText.charCodeAt(0);
194|       const second = fullText.charCodeAt(1);
195|       cut = (first >= 0xd800 && first <= 0xdbff && second >= 0xdc00 && second <= 0xdfff) ? 2 : 1;
196|     }
197|     const text = fullText.slice(0, cut);
198|     if (fullText.length > maxCharsPerMessage) textTruncated = true;
199|     const content = nodeContent(node);
200|     const meta = collectSegmentMeta(content);
201|     return {
202|       index: index + 1,
203|       sender: senderName(node),
204|       userId: senderUserId(node),
205|       time: nodeTime(node),
206|       messageId: nodeMessageId(node),
207|       messageSeq: nodeMessageSeq(node),
208|       text,
209|       media: meta.media,
210|       nestedForwardIds: meta.nestedForwardIds
211|     };
212|   });
213|   return {
214|     total,
215|     truncated: truncatedByCount || textTruncated,
216|     textTruncated,
217|     messages
218|   };
