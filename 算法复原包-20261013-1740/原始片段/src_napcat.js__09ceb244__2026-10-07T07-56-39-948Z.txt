import { spawn, execFile } from "node:child_process";
import fs from "node:fs";
import path from "node:path";

export function createNapcatLauncher(ctx) {
let proc = null;
let startedAt = null;

function cfg() { return ctx.config.napcat || {}; }

let lastCheck = 0;
let lastCount = 0;

function procByName(name, force) {
// 5 秒缓存：状态轮询很频繁，避免每次都起 PowerShell
if (!force && Date.now() - lastCheck < 5000) return Promise.resolve(lastCount);
return new Promise(function (resolve) {
const ps = "C:\\Windows\\System32\\WindowsPowerShell\\v1.0\\powershell.exe";
const cmd = "(Get-Process -Name '" + name + "' -ErrorAction SilentlyContinue | Measure-Object).Count";
execFile(ps, ["-NoProfile", "-Command", cmd], { windowsHide: true, timeout: 15000 }, function (err, stdout) {
const n = parseInt(String(stdout || "0").trim(), 10);
lastCheck = Date.now();
lastCount = isNaN(n) ? 0 : n;
resolve(lastCount);
});
});
}

async function status() {
const c = cfg();
let alive = false;
try { alive = !!(proc && proc.exitCode === null && !proc.killed); } catch (e) { alive = false; }
if (!alive) {
try { alive = (await procByName("NapCatWinBootMain")) > 0; } catch (e) { alive = false; }
}
return {
configured: !!(c.launcherPath && String(c.launcherPath).trim()),
launcherPath: c.launcherPath || "",
autoStart: !!c.autoStart,
hideConsole: c.hideConsole !== false,
qq: c.qq || "",
running: alive,
pid: alive ? proc.pid : null,
startedAt: alive ? startedAt : null
};
}

async function start(options) {
const forceShow = !!(options && options.show);
const c = cfg();
const p = String(c.launcherPath || "").trim();
if (!p) return { ok: false, error: "未配置 NapCat 启动路径" };
if (!fs.existsSync(p)) return { ok: false, error: "路径不存在：" + p };
try { if (proc && proc.exitCode === null && !proc.killed) return { ok: true, already: true, pid: proc.pid }; } catch (e) {}
// 已经有一个 NapCat 在跑（可能是上次留下的），绝不再拉一个，否则两个实例会抢登录态和端口
try {
const running = await procByName("NapCatWinBootMain", true);
if (running > 0) return { ok: true, already: true, pid: null, note: "检测到已有 NapCat 在运行" };
} catch (e) {}
const args = [];
const qq = String(c.qq || "").trim();
if (qq) args.push(qq);
// 注意：托盘启动的 node 自身没有控制台，无法为子进程创建可见窗口。
// 需要看窗口（首次扫码）时请用托盘菜单的「启动 NapCat（显示窗口）」。
const cwd = path.dirname(p);
try {
const opts = { cwd: cwd, detached: true, stdio: "ignore", windowsHide: true };
proc = /\.(bat|cmd)$/i.test(p) ? spawn("cmd.exe", ["/c", p].concat(args), opts) : spawn(p, args, opts);
proc.unref();
startedAt = Date.now();
return { ok: true, pid: proc.pid, hidden: true };
} catch (e) {
return { ok: false, error: "启动失败：" + String((e && e.message) || e) };
}
}

function killNapcatQQ() {
return new Promise(function (resolve) {
const ps = "C:\\Windows\\System32\\WindowsPowerShell\\v1.0\\powershell.exe";
const cmd = "Get-CimInstance Win32_Process -Filter \"Name='QQ.exe'\" | Where-Object { $_.CommandLine -like '*--enable-logging*' } | ForEach-Object { $_.ProcessId }";
execFile(ps, ["-NoProfile", "-Command", cmd], { windowsHide: true, timeout: 20000 }, function (err, stdout) {
const pids = String(stdout || "").split(/\s+/).filter(function (s) { return /^[0-9]+$/.test(s); });
const killed = [];
for (const p of pids) {
try { process.kill(Number(p)); killed.push(Number(p)); } catch (e) {}
}
resolve(killed);
});
});
}

async function stop() {
const killed = [];
try { if (proc && proc.exitCode === null && !proc.killed) { proc.kill(); killed.push("launcher"); } } catch (e) {}
proc = null;
startedAt = null;
let qq = [];
try { qq = await killNapcatQQ(); } catch (e) { qq = []; }
return { ok: true, killedLauncher: killed.length > 0, killedQQ: qq };
}

return { start: start, stop: stop, status: status };
}
