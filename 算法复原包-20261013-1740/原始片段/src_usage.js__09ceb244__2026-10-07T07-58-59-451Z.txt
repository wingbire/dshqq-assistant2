import fs from "node:fs";
import path from "node:path";
import { DATA_ROOT, ensureDir } from "./paths.js";

const FILE = path.join(DATA_ROOT, "data", "usage.json");

// DeepSeek 公开价（元 / 百万 token）。高峰：工作日 9-12、14-18；其余为空闲。
const PRICE = {
offPeak: { cacheHit: 0.02, cacheMiss: 1.0, output: 4.0 },
peak: { cacheHit: 0.04, cacheMiss: 2.0, output: 8.0 }
};

export function isPeak(date) {
const d = date || new Date();
const day = d.getDay();
if (day === 0 || day === 6) return false;
const h = d.getHours();
return (h >= 9 && h < 12) || (h >= 14 && h < 18);
}

export function costOf(usage, peak) {
const p = peak ? PRICE.peak : PRICE.offPeak;
const hit = Number(usage.cacheReadTokens || 0);
const miss = Number(usage.inputTokens || 0);
const out = Number(usage.outputTokens || 0);
return (hit * p.cacheHit + miss * p.cacheMiss + out * p.output) / 1000000;
}

// 「计费 token」= 未命中缓存的输入 + 输出。缓存读只有 ¥0.02/百万，
// 却会被 context 重放放大到总量的八九成，拿它当预算口径会让预算形同虚设。
export function billableOf(usage) {
if (!usage) return 0;
return Number(usage.inputTokens || 0) + Number(usage.outputTokens || 0);
}

function today(date) {
const d = date || new Date();
const p = (n) => (n < 10 ? "0" + n : "" + n);
return d.getFullYear() + "-" + p(d.getMonth() + 1) + "-" + p(d.getDate());
}

function emptyBucket() {
return { calls: 0, inputTokens: 0, cacheReadTokens: 0, outputTokens: 0, totalTokens: 0, billableTokens: 0, cost: 0, byGroup: {} };
}

export function createUsage() {
let data = load();

function load() {
try { const o = JSON.parse(fs.readFileSync(FILE, "utf8")); return o && o.days ? o : { days: {} }; }
catch (e) { return { days: {} }; }
}

function save() {
ensureDir(path.dirname(FILE));
const keys = Object.keys(data.days).sort();
while (keys.length > 15) { const k = keys.shift(); delete data.days[k]; }   // 只保留最近 15 天
const tmp = FILE + ".tmp";
fs.writeFileSync(tmp, JSON.stringify(data, null, 2), "utf8");
fs.renameSync(tmp, FILE);
}

const day = today();
const peak = isPeak();
const cost = usage ? costOf(usage, peak) : 0;

const bucket = d.byGroup[g] || (d.byGroup[g] = { calls: 0, inputTokens: 0, cacheReadTokens: 0, outputTokens: 0, totalTokens: 0, billableTokens: 0, cost: 0, cached: 0, merged: 0, deduped: 0 });
if (usage) {
const billable = billableOf(usage);
for (const target of [d, bucket]) {
target.calls += 1;
target.inputTokens += Number(usage.inputTokens || 0);
target.cacheReadTokens += Number(usage.cacheReadTokens || 0);
target.outputTokens += Number(usage.outputTokens || 0);
target.totalTokens += Number(usage.totalTokens || 0);
target.billableTokens = Number(target.billableTokens || 0) + billable;
target.cost += cost;
}
}
if (extra && extra.deduped) { d.deduped = (d.deduped || 0) + 1; bucket.deduped += 1; }
if (extra && extra.merged) { d.merged = (d.merged || 0) + 1; bucket.merged += 1; }
save();
return cost;
}

// 每日预算口径：只看计费 token（老数据没有该字段则按 0 起算）
function groupTodayTokens(gid) {
const d = data.days[today()];
if (!d) return 0;
const b = d.byGroup[String(gid)];
return b ? Number(b.billableTokens || 0) : 0;
}

function groupTodayTotalTokens(gid) {
const d = data.days[today()];
if (!d) return 0;
const b = d.byGroup[String(gid)];
return b ? Number(b.totalTokens || 0) : 0;
}

function todayTotals() {
const d = data.days[today()] || emptyBucket();
const hitRate = d.cacheReadTokens + d.inputTokens > 0 ? d.cacheReadTokens / (d.cacheReadTokens + d.inputTokens) : 0;
return {
day: today(),
peak: isPeak(),
calls: d.calls || 0,
inputTokens: d.inputTokens || 0,
cacheReadTokens: d.cacheReadTokens || 0,
outputTokens: d.outputTokens || 0,
totalTokens: d.totalTokens || 0,
billableTokens: Number(d.billableTokens || 0),
cost: d.cost || 0,
deduped: d.deduped || 0,
merged: d.merged || 0,
cacheHitRate: hitRate,
byGroup: (d.byGroup || {})
};
}

function history(days) {
const out = [];
for (const k of Object.keys(data.days).sort().reverse().slice(0, days || 15)) {
const d = data.days[k];
out.push({ day: k, calls: d.calls || 0, totalTokens: d.totalTokens || 0, cost: d.cost || 0, deduped: d.deduped || 0 });
}
return out;
}

function resetToday() {
delete data.days[today()];
save();
}

return {
record: record,
groupTodayTokens: groupTodayTokens,
groupTodayTotalTokens: groupTodayTotalTokens,
todayTotals: todayTotals,
history: history,
resetToday: resetToday,
price: PRICE
};
}
